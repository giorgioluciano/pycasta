#!/usr/bin/env python
# -*- coding: utf-8 -*-

from collections import defaultdict
import numpy as np


def build_alpha_tetra_adjacency(simplices, alpha_mask):
    simplices = np.asarray(simplices, dtype=int)
    alpha_mask = np.asarray(alpha_mask, dtype=bool)

    alpha_indices = np.where(alpha_mask)[0]
    alpha_set = set(alpha_indices.tolist())

    face_to_tetra = defaultdict(list)
    for tetra_idx in alpha_indices:
        tet = simplices[tetra_idx]
        faces = [
            tuple(sorted((tet[0], tet[1], tet[2]))),
            tuple(sorted((tet[0], tet[1], tet[3]))),
            tuple(sorted((tet[0], tet[2], tet[3]))),
            tuple(sorted((tet[1], tet[2], tet[3]))),
        ]
        for face in faces:
            face_to_tetra[face].append(int(tetra_idx))

    graph = {int(i): set() for i in alpha_indices}
    for tetra_list in face_to_tetra.values():
        if len(tetra_list) < 2:
            continue
        for i in range(len(tetra_list)):
            for j in range(i + 1, len(tetra_list)):
                a = tetra_list[i]
                b = tetra_list[j]
                if a in alpha_set and b in alpha_set:
                    graph[a].add(b)
                    graph[b].add(a)

    return graph


def connected_components_from_graph(graph):
    visited = set()
    components = []

    for node in graph:
        if node in visited:
            continue
        stack = [node]
        component = []

        while stack:
            cur = stack.pop()
            if cur in visited:
                continue
            visited.add(cur)
            component.append(cur)
            stack.extend(n for n in graph[cur] if n not in visited)

        components.append(sorted(component))

    components.sort(key=len, reverse=True)
    return components


def compute_alpha_connectivity_metrics(simplices, alpha_mask):
    alpha_mask = np.asarray(alpha_mask, dtype=bool)
    num_alpha_tetrahedra = int(alpha_mask.sum())

    if num_alpha_tetrahedra == 0:
        return {
            "num_alpha_components": 0,
            "largest_component_size": 0,
            "largest_component_ratio": 0.0,
            "small_component_fraction": 0.0,
            "component_size_entropy": 0.0,
        }

    graph = build_alpha_tetra_adjacency(simplices, alpha_mask)
    components = connected_components_from_graph(graph)

    component_sizes = np.array([len(c) for c in components], dtype=float)
    largest_component_size = int(component_sizes.max()) if component_sizes.size else 0
    largest_component_ratio = (
        float(largest_component_size / num_alpha_tetrahedra)
        if num_alpha_tetrahedra > 0
        else 0.0
    )

    if component_sizes.size <= 1:
        small_component_fraction = 0.0
        component_size_entropy = 0.0
    else:
        small_component_fraction = float(
            component_sizes[1:].sum() / num_alpha_tetrahedra
        )
        probs = component_sizes / component_sizes.sum()
        component_size_entropy = float(
            -np.sum(probs * np.log(probs + 1e-12))
        )

    return {
        "num_alpha_components": int(len(components)),
        "largest_component_size": largest_component_size,
        "largest_component_ratio": largest_component_ratio,
        "small_component_fraction": small_component_fraction,
        "component_size_entropy": component_size_entropy,
    }

#!/usr/bin/env python
# -*- coding: utf-8 -*-

import numpy as np
from scipy.spatial import KDTree


def compute_tetra_max_edge_lengths(tetra_positions):
    tetra_positions = np.asarray(tetra_positions, dtype=float)

    if tetra_positions.ndim != 3 or tetra_positions.shape[1:] != (4, 3):
        raise ValueError(
            f"Expected tetra_positions shape (n, 4, 3), got {tetra_positions.shape}"
        )

    edge_pairs = [
        (0, 1),
        (0, 2),
        (0, 3),
        (1, 2),
        (1, 3),
        (2, 3),
    ]

    edge_lengths = []
    for i, j in edge_pairs:
        edge_lengths.append(
            np.linalg.norm(tetra_positions[:, i, :] - tetra_positions[:, j, :], axis=1)
        )

    edge_lengths = np.stack(edge_lengths, axis=1)
    max_edge_lengths = np.max(edge_lengths, axis=1)
    return max_edge_lengths


def compute_local_atom_scale(protein_coords, k=6):
    protein_coords = np.asarray(protein_coords, dtype=float)

    if protein_coords.ndim != 2 or protein_coords.shape[1] != 3:
        raise ValueError(
            f"Expected protein_coords shape (n, 3), got {protein_coords.shape}"
        )

    if len(protein_coords) < 2:
        return np.ones(len(protein_coords), dtype=float)

    k_eff = min(max(int(k), 2), len(protein_coords))
    tree = KDTree(protein_coords)
    dists, _ = tree.query(protein_coords, k=k_eff)

    if dists.ndim == 1:
        dists = dists[:, None]

    # skip self-distance in column 0
    if dists.shape[1] > 1:
        local_scale = np.mean(dists[:, 1:], axis=1)
    else:
        local_scale = np.ones(len(protein_coords), dtype=float)

    local_scale = np.where(local_scale > 1e-8, local_scale, 1.0)
    return local_scale


def compute_tetra_local_scale(tetra_positions, protein_coords, k=6):
    tetra_positions = np.asarray(tetra_positions, dtype=float)
    protein_coords = np.asarray(protein_coords, dtype=float)

    atom_local_scale = compute_local_atom_scale(protein_coords, k=k)

    tree = KDTree(protein_coords)
    flat_vertices = tetra_positions.reshape(-1, 3)
    _, nearest_idx = tree.query(flat_vertices, k=1)

    vertex_scale = atom_local_scale[np.asarray(nearest_idx, dtype=int)]
    vertex_scale = vertex_scale.reshape(-1, 4)

    tetra_local_scale = np.mean(vertex_scale, axis=1)
    tetra_local_scale = np.where(tetra_local_scale > 1e-8, tetra_local_scale, 1.0)
    return tetra_local_scale


def compute_tetra_edge_ratio(
    tetra_positions,
    protein_coords,
    local_scale_k=6,
):
    max_edge_lengths = compute_tetra_max_edge_lengths(tetra_positions)
    tetra_local_scale = compute_tetra_local_scale(
        tetra_positions=tetra_positions,
        protein_coords=protein_coords,
        k=local_scale_k,
    )

    edge_ratio = max_edge_lengths / tetra_local_scale
    return edge_ratio, max_edge_lengths, tetra_local_scale


def build_graph_constraint_mask(
    tetra_positions,
    protein_coords,
    max_edge_factor=2.5,
    local_scale_k=6,
):
    edge_ratio, max_edge_lengths, tetra_local_scale = compute_tetra_edge_ratio(
        tetra_positions=tetra_positions,
        protein_coords=protein_coords,
        local_scale_k=local_scale_k,
    )

    constraint_mask = edge_ratio <= float(max_edge_factor)

    stats = {
        "num_total_tetrahedra": int(len(edge_ratio)),
        "num_kept_by_graph_filter": int(np.sum(constraint_mask)),
        "num_removed_by_graph_filter": int(np.sum(~constraint_mask)),
        "kept_fraction_by_graph_filter": (
            float(np.mean(constraint_mask)) if len(constraint_mask) > 0 else 0.0
        ),
        "edge_ratio_mean": float(np.mean(edge_ratio)) if len(edge_ratio) > 0 else 0.0,
        "edge_ratio_median": float(np.median(edge_ratio)) if len(edge_ratio) > 0 else 0.0,
        "edge_ratio_max": float(np.max(edge_ratio)) if len(edge_ratio) > 0 else 0.0,
        "max_edge_length_mean": (
            float(np.mean(max_edge_lengths)) if len(max_edge_lengths) > 0 else 0.0
        ),
        "local_scale_mean": (
            float(np.mean(tetra_local_scale)) if len(tetra_local_scale) > 0 else 0.0
        ),
        "max_edge_factor": float(max_edge_factor),
        "local_scale_k": int(local_scale_k),
    }

    return constraint_mask, stats


def apply_alpha_constraints(
    alpha_mask,
    tetra_positions,
    protein_coords,
    use_graph_filter=True,
    max_edge_factor=2.5,
    local_scale_k=6,
):
    alpha_mask = np.asarray(alpha_mask, dtype=bool)

    if not use_graph_filter:
        stats = {
            "graph_filter_enabled": False,
            "num_alpha_before_graph_filter": int(np.sum(alpha_mask)),
            "num_alpha_after_graph_filter": int(np.sum(alpha_mask)),
            "num_removed_from_alpha_by_graph_filter": 0,
        }
        return alpha_mask, stats

    constraint_mask, constraint_stats = build_graph_constraint_mask(
        tetra_positions=tetra_positions,
        protein_coords=protein_coords,
        max_edge_factor=max_edge_factor,
        local_scale_k=local_scale_k,
    )

    constrained_alpha_mask = alpha_mask & constraint_mask

    stats = {
        "graph_filter_enabled": True,
        "num_alpha_before_graph_filter": int(np.sum(alpha_mask)),
        "num_alpha_after_graph_filter": int(np.sum(constrained_alpha_mask)),
        "num_removed_from_alpha_by_graph_filter": int(
            np.sum(alpha_mask & (~constraint_mask))
        ),
    }
    stats.update(constraint_stats)

    return constrained_alpha_mask, stats

def _alpha_grid_from_base(base_alpha, radii, alpha_values=None, alpha_offsets=None):
    radii = np.asarray(radii, dtype=float)

    if alpha_values is not None:
        vals = sorted({round(float(a), 4) for a in alpha_values if float(a) > 0})
        return vals

    if alpha_offsets is None:
        alpha_offsets = [-0.3, -0.2, -0.1, 0.0, 0.1, 0.2, 0.3]

    vals = sorted(
        {
            round(float(base_alpha + off), 4)
            for off in alpha_offsets
            if float(base_alpha + off) > 0
        }
    )

    if radii.size > 0:
        q25 = float(np.quantile(radii, 0.25))
        q50 = float(np.quantile(radii, 0.50))
        q75 = float(np.quantile(radii, 0.75))
        vals = sorted(set(vals + [round(q25, 4), round(q50, 4), round(q75, 4)]))

    return vals

    pockets, flow_steps, connectivity_graph = flow_detection(
        simplices,
        alpha_mask,
        tetra_positions,
        protein_coords,
        flow_params,
        proxy_values=radii,  # radii is already computed earlier in this function
    )

    max_index = len(protein_coords) - 1
    valid_pockets = [p for p in pockets if all(0 <= idx <= max_index for idx in p)]
    num_preliminary_pockets = len(valid_pockets)
    max_index = len(protein_coords) - 1
    valid_pockets = [p for p in pockets if all(0 <= idx <= max_index for idx in p)]
    num_preliminary_pockets = len(valid_pockets)

    if merge_clusters:
        merged_pockets = merge_nearby_clusters(
            valid_pockets,
            local_tetra_positions,
            merge_threshold,
        )
    else:
        merged_pockets = valid_pockets

    num_merged_pockets = len(merged_pockets)

    pocket_volumes = [
        compute_analytic_pocket_volume(local_tetra_positions, pocket)
        for pocket in merged_pockets
    ]

    filtered_volumes = [
        float(v) for v in pocket_volumes if float(v) >= float(min_volume_threshold)
    ]
    filtered_volumes = sorted(filtered_volumes, reverse=True)

    flow_values = list(flow_steps.values())
    mean_flow_steps = float(np.mean(flow_values)) if flow_values else 0.0
    max_flow_steps = int(np.max(flow_values)) if flow_values else 0

    return {
        "alpha": float(alpha_value),
        "total_tetrahedra": total_tetra,
        "num_alpha_tetrahedra": alpha_tetra,
        "alpha_fraction": alpha_fraction,
        "graph_filter_enabled": bool(
            graph_constraint_stats.get("graph_filter_enabled", False)
        ),
        "num_alpha_before_graph_filter": int(
            graph_constraint_stats.get("num_alpha_before_graph_filter", alpha_tetra)
        ),
        "num_alpha_after_graph_filter": int(
            graph_constraint_stats.get("num_alpha_after_graph_filter", alpha_tetra)
        ),
        "num_removed_from_alpha_by_graph_filter": int(
            graph_constraint_stats.get("num_removed_from_alpha_by_graph_filter", 0)
        ),
        "kept_fraction_by_graph_filter": float(
            graph_constraint_stats.get("kept_fraction_by_graph_filter", 0.0)
        ),
        "edge_ratio_mean": float(graph_constraint_stats.get("edge_ratio_mean", 0.0)),
        "edge_ratio_median": float(
            graph_constraint_stats.get("edge_ratio_median", 0.0)
        ),
        "edge_ratio_max": float(graph_constraint_stats.get("edge_ratio_max", 0.0)),
        "max_edge_length_mean": float(
            graph_constraint_stats.get("max_edge_length_mean", 0.0)
        ),
        "local_scale_mean": float(
            graph_constraint_stats.get("local_scale_mean", 0.0)
        ),
        "num_preliminary_pockets": num_preliminary_pockets,
        "num_merged_pockets": num_merged_pockets,
        "num_final_pockets": len(filtered_volumes),
        "mean_flow_steps": mean_flow_steps,
        "max_flow_steps": max_flow_steps,
        "mean_volume": float(np.mean(filtered_volumes)) if filtered_volumes else 0.0,
        "median_volume": float(np.median(filtered_volumes)) if filtered_volumes else 0.0,
        "max_volume": float(np.max(filtered_volumes)) if filtered_volumes else 0.0,
        "top3_volumes": [round(v, 4) for v in filtered_volumes[:3]],
        "all_volumes": [round(v, 4) for v in filtered_volumes],
    }


def _save_radii_histogram(radii, alpha_values, out_png, title):
    radii = np.asarray(radii, dtype=float)

    plt.figure(figsize=(10, 6))
    if radii.size > 0:
        plt.hist(radii, bins=50, color="#4C78A8", alpha=0.8, edgecolor="white")
        for alpha in alpha_values:
            plt.axvline(alpha, linestyle="--", linewidth=1.5, label=f"alpha={alpha:.2f}")
    plt.xlabel("Circumsphere radius")
    plt.ylabel("Count")
    plt.title(title)
    if len(alpha_values) <= 10:
        plt.legend()
    plt.tight_layout()
    plt.savefig(out_png, dpi=180)
    plt.close()

            "top3_png": top3_png,
        },
    }

    with open(json_path, "w", encoding="utf-8") as f:
        json.dump(summary, f, indent=2)

    return summary

import json

def show_alpha_report(pdb_name, results_dir, display_pngs=True):
    results = Path(results_dir)
    csvs = sorted(results.glob('*alphasweepmetrics.csv'))
    jsons = sorted(results.glob('*alphadiagnostics.json'))
    if not csvs or not jsons:
        raise FileNotFoundError('Alpha diagnostics artifacts not found')
    csv_path, json_path = csvs[0], jsons[0]
    df = pd.read_csv(csv_path)
    with open(json_path, 'r', encoding='utf-8') as f:
        summary = json.load(f)
    display(Markdown(f"## Alpha report — {pdb_name}"))
    display(Markdown(f"**Base alpha:** {summary.get('basealpha')}  
**Tested alphas:** {summary.get('testedalphas')}"))
    cols = [c for c in ['alpha','numalphatetrahedra','numpreliminarypockets','nummergedpockets','numfinalpockets','meanflowsteps','maxvolume'] if c in df.columns]
    display(df[cols].sort_values('alpha'))
    if display_pngs:
        for key in ['histpng','sweeppng','top3png']:
            p = summary.get('artifacts', {}).get(key)
            if p:
                display(Image(filename=p))
    return df, summary

def filter_alpha_complex_by_sasa(
    tetrahedra, alpha_values, protein_coords, pdb_file, sasa_threshold
):
    logging.info("Computing SASA for filtering the alpha complex...")
    sasa_dict = compute_sasa(pdb_file, method="pymol", remove_ligand=True)
    if sasa_dict is None:
        logging.error("SASA computation failed; no filtering applied.")
        return tetrahedra, alpha_values, np.ones_like(alpha_values, dtype=bool)
    ppdb = PandasPdb().read_pdb(pdb_file)
    atoms_df = ppdb.df["ATOM"]
    atom_residues = atoms_df["residue_number"].values
    

    tree = KDTree(protein_coords)
    filtered_indices = []
    total_tetra = len(alpha_values)
    for i, tet in enumerate(tetrahedra):
        centroid = np.mean(tet, axis=0)
        dist, idx = tree.query(centroid)
        residue = atom_residues[idx]
        sasa = sasa_dict.get(str(residue), 0)
        if sasa < sasa_threshold:
            filtered_indices.append(i)
    logging.info(
        f"Tetrahedra: {total_tetra} total, {len(filtered_indices)} pass the SASA filter (threshold={sasa_threshold})."
    )
    filtered_tetrahedra = tetrahedra[filtered_indices]
    filtered_alpha_values = alpha_values[filtered_indices]
    filtered_mask = np.zeros_like(alpha_values, dtype=bool)
    filtered_mask[filtered_indices] = True
    return filtered_tetrahedra, filtered_alpha_values, filtered_mask

def compute_alpha_complex_from_tetrahedra(
    simplices,
    tetra_positions,
    alpha_value,
    molecule_name=None,
    protein_coords=None,
    radii=None,  # NEW: atomic radii for weighted Delaunay (power radii)
    filter_by_sasa=False,
    pdb_file=None,
    save_alpha_data=True,
):
    """
    Compute the alpha complex for a set of tetrahedra.
    
    This function supports both standard (unweighted) and weighted (regular) Delaunay 
    triangulations:
    
    
    
    All existing features are preserved:
    - Graph-based alpha constraints (USE_GRAPH_FILTERED_ALPHA)
    - SASA filtering (filter_by_sasa)
    - Debug data saving (SAVE_TRIANGULATION_DEBUG, SAVE_ALPHA)
    
    Parameters
    ----------
    simplices : np.ndarray
        Array of shape (N, 4) containing vertex indices for each tetrahedron 
        in the original triangulation.
    tetra_positions : np.ndarray
        Array of shape (N, 4, 3) containing 3D coordinates of the 4 vertices 
        for each tetrahedron.
    alpha_value : float
        Alpha threshold for filtering tetrahedra (tetrahedra with radius < alpha_value are kept).
    molecule_name : str, optional
        Name of the molecule (used for logging and file naming).
    protein_coords : np.ndarray, optional
        Array of shape (M, 3) containing coordinates of all atoms in the protein.
        Required for weighted Delaunay (power radii) and graph-based constraints.
    radii : np.ndarray, optional
        Array of shape (M,) containing atomic radii (e.g., van der Waals radii).
        Required for weighted Delaunay (power radii). If None, falls back to 
        standard circumradii.
    filter_by_sasa : bool, optional
        If True, apply SASA-based filtering to the alpha complex.
    sasa_threshold : float, optional
        SASA threshold for filtering (default: SASA_CONTACT_THRESHOLD).
    pdb_file : str, optional
        Path to the PDB file (required for SASA filtering).
    save_alpha_data : bool, optional
        If True, save alpha shape data to NPZ files for debugging.
    
    Returns
    -------
    alpha_mask : np.ndarray
        Boolean mask of shape (N,) indicating which tetrahedra belong to the 
        alpha complex (radius < alpha_value).
    radii : np.ndarray
        Array of shape (N,) containing the radii used for filtering 
        (either circumradii or power radii).
    tetra_positions : np.ndarray
        Possibly updated tetra_positions after SASA filtering.
    simplices : np.ndarray
        Filtered simplices array (aligned with tetra_positions after SASA filtering).
    """
    import os
    import logging
    import numpy as np
    from .config import (
        SAVE_TRIANGULATION_DEBUG,
        SAVE_ALPHA,
        OUTPUT_DIR,
        VERSION_TAG,
        USE_GRAPH_FILTERED_ALPHA,
        ALPHA_MAX_EDGE_FACTOR,
        ALPHA_LOCAL_SCALE_K,
    )
    from .utils.geometry_utils import compute_all_circumsphere_radii, compute_power_radii, log_radii_statistics
    from .alpha_constraints import apply_alpha_constraints
    #from .utils.sasa_utils import filter_alpha_complex_by_sasa
    
    # Validate input shape
    assert tetra_positions.ndim == 3 and tetra_positions.shape[1:] == (
        4,
        3,
    ), f"Invalid tetra_positions shape: {tetra_positions.shape}"
    
    # =========================================================================
    # STEP 1: Compute radii (circumradii or power radii)
    # =========================================================================
    

   

    protein_coords = np.asarray(protein_coords, dtype=float)
    radii_array = np.asarray(radii, dtype=float)
       
    radii = compute_power_radii(simplices, protein_coords, radii_array)
    
    log_radii_statistics(radii)
    
    # Compute initial alpha mask based on radius threshold
    alpha_mask = (radii < alpha_value).astype(bool)
    
    # =========================================================================
    # STEP 2: Save raw triangulation debug data (before alpha filtering)
    # =========================================================================
    
    if SAVE_TRIANGULATION_DEBUG:
        try:
            save_triangulation_debug_data(
                simplices=simplices,
                tetra_positions=tetra_positions,
                protein_coords=protein_coords,
                molecule_name=molecule_name,
                radii=radii,
            )
        except Exception as e:
            logging.warning(f"[triangulation_debug] failed to save: {e}")
    
    # =========================================================================
    # STEP 3: Apply graph-based alpha constraints (if enabled)
    # =========================================================================
    
    constraint_stats = {
        "graph_filter_enabled": False,
        "num_alpha_before_graph_filter": int(np.sum(alpha_mask)),
        "num_alpha_after_graph_filter": int(np.sum(alpha_mask)),
        "num_removed_from_alpha_by_graph_filter": 0,
    }
    
    if USE_GRAPH_FILTERED_ALPHA:
        if protein_coords is None:
            logging.warning(
                "USE_GRAPH_FILTERED_ALPHA=True but protein_coords is None; "
                "skipping graph-based alpha constraints."
            )
        else:
            alpha_mask, constraint_stats = apply_alpha_constraints(
                alpha_mask=alpha_mask,
                tetra_positions=tetra_positions,
                protein_coords=protein_coords,
                use_graph_filter=True,
                max_edge_factor=ALPHA_MAX_EDGE_FACTOR,
                local_scale_k=ALPHA_LOCAL_SCALE_K,
            )
            
            logging.info(
                "[alpha_constraints] before=%d after=%d removed=%d edge_factor=%.3f k=%d",
                constraint_stats["num_alpha_before_graph_filter"],
                constraint_stats["num_alpha_after_graph_filter"],
                constraint_stats["num_removed_from_alpha_by_graph_filter"],
                ALPHA_MAX_EDGE_FACTOR,
                ALPHA_LOCAL_SCALE_K,
            )
    
    # =========================================================================
    # STEP 4: Save alpha shape data (before SASA filtering)
    # =========================================================================
    
    if save_alpha_data and SAVE_ALPHA and molecule_name and protein_coords is not None:
        version_dir = os.path.join(OUTPUT_DIR, VERSION_TAG)
        os.makedirs(version_dir, exist_ok=True)
        
        out_file = os.path.join(version_dir, f"{molecule_name}.alpha.npz")
        
        np.savez(
            out_file,
            alpha_value=float(alpha_value),
            alpha_values=radii,
            alpha_mask=alpha_mask,
            protein_coords=protein_coords,
            tetrahedra=tetra_positions,
        )
        
        logging.info(f"Alpha shape data saved to {out_file}")
    
    # =========================================================================
    # STEP 5: Apply SASA filtering (if enabled)
    # =========================================================================
    
    if filter_by_sasa:
        if pdb_file is None:
            logging.error("pdb_file must be provided when filtering by SASA.")
        else:
            # Filter alpha complex by SASA
            filtered_tetra, filtered_alpha_values, filtered_mask = (
                filter_alpha_complex_by_sasa(
                    tetra_positions, radii, protein_coords, pdb_file, sasa_threshold
                )
            )
            
            # Update tetra_positions and radii based on the SASA filter
            tetra_positions = filtered_tetra
            radii = filtered_alpha_values
            
            # Recalculate alpha_mask from the filtered radii
            alpha_mask = (radii < alpha_value).astype(bool)
            
            # Reset constraint stats for post-SASA filtering
            constraint_stats = {
                "graph_filter_enabled": False,
                "num_alpha_before_graph_filter": int(np.sum(alpha_mask)),
                "num_alpha_after_graph_filter": int(np.sum(alpha_mask)),
                "num_removed_from_alpha_by_graph_filter": 0,
            }
            
            # Re-apply graph-based constraints after SASA filtering
            if USE_GRAPH_FILTERED_ALPHA:
                if protein_coords is None:
                    logging.warning(
                        "USE_GRAPH_FILTERED_ALPHA=True but protein_coords is None; "
                        "skipping graph-based alpha constraints after SASA filtering."
                    )
                else:
                    alpha_mask, constraint_stats = apply_alpha_constraints(
                        alpha_mask=alpha_mask,
                        tetra_positions=tetra_positions,
                        protein_coords=protein_coords,
                        use_graph_filter=True,
                        max_edge_factor=ALPHA_MAX_EDGE_FACTOR,
                        local_scale_k=ALPHA_LOCAL_SCALE_K,
                    )
                    
                    logging.info(
                        "[alpha_constraints][post_sasa] before=%d after=%d removed=%d edge_factor=%.3f k=%d",
                        constraint_stats["num_alpha_before_graph_filter"],
                        constraint_stats["num_alpha_after_graph_filter"],
                        constraint_stats["num_removed_from_alpha_by_graph_filter"],
                        ALPHA_MAX_EDGE_FACTOR,
                        ALPHA_LOCAL_SCALE_K,
                    )
            
            # Update simplices using the same boolean mask from SASA filtering
            simplices = simplices[filtered_mask]
            
            # Save filtered alpha shape data
            version_dir = os.path.join(OUTPUT_DIR, VERSION_TAG)
            os.makedirs(version_dir, exist_ok=True)
            out_file_filtered = os.path.join(
                version_dir, f"{molecule_name}.alpha_filtered.npz"
            )
            
            np.savez(
                out_file_filtered,
                alpha_values=radii,
                alpha_mask=alpha_mask,
                protein_coords=protein_coords,
                tetrahedra=tetra_positions,
            )
            logging.info(f"Filtered alpha shape data saved to {out_file_filtered}")
    
    # =========================================================================
    # STEP 6: Return results
    # =========================================================================
    
    return alpha_mask, radii, tetra_positions, simplices




def flow_detection(
    simplices,
    alpha_mask,
    tetra_positions,
    protein_coords,
    flow_params=None,
    proxy_values=None,  # NEW: precomputed radii (circum or power)
):
    """
    Apply discrete flow on the tetrahedra to grow pockets.
    This version uses a relative tolerance (tol_fraction) to decide if a neighbor is lower.
    
    Parameters
    ----------
    proxy_values : np.ndarray, optional
        Precomputed proxy values (e.g., circumradii or power radii) for all tetrahedra.
        If None, defaults to computing circumradii from tetra_positions.
    """
    if flow_params is None:
        flow_params = {}
    sigma_p = flow_params.get("sigma_p", 1.0)
    # Use a relative tolerance fraction (e.g., 0.01 means a 1% drop)
    tol_fraction = flow_params.get("tol_fraction", 0.01)
    max_steps = flow_params.get("max_steps", 100)
    adaptive = flow_params.get("adaptive", False)
    adaptive_factor = flow_params.get("adaptive_factor", 0.9)
    min_steps = flow_params.get("min_steps", 0)

    # Compute proxy values for all tetrahedra (e.g., based on circumsphere radii)
    if proxy_values is None:
        # Fallback: compute circumradii (classic mode)
        proxies = compute_all_circumsphere_radii(tetra_positions)
        proxy_values = sigma_p * proxies
    else:
        # Use precomputed proxy values (e.g., power radii from weighted Delaunay)
        # sigma_p is already applied if needed by the caller
        proxy_values = np.asarray(proxy_values, dtype=float)

    # Get indices of tetrahedra that are outside the alpha complex
    empty_indices = np.where(~alpha_mask)[0]

    # Build a face-to-tetrahedron mapping (neighbors sharing a face)
    face_to_tetra = {}
    for i, simplex in enumerate(simplices):
        faces = [
            tuple(sorted(simplex[[a, b, c]]))
            for (a, b, c) in [(0, 1, 2), (0, 1, 3), (0, 2, 3), (1, 2, 3)]
        ]
        for face in faces:
            face_to_tetra.setdefault(face, []).append(i)

    # Build neighbor dictionary for each empty tetrahedron
    neighbor_dict = {}
    for i in empty_indices:
        faces = [
            tuple(sorted(simplices[i, [a, b, c]]))
            for (a, b, c) in [(0, 1, 2), (0, 1, 3), (0, 2, 3), (1, 2, 3)]
        ]
        neighbors = set()
        for face in faces:
            neighbors.update(face_to_tetra.get(face, []))
        # Exclude the tetrahedron itself and those that are not in the empty region
        neighbor_dict[i] = [j for j in neighbors if j != i and not alpha_mask[j]]

    # Flow: for each tetrahedron in empty_indices, try to follow a gradient descent
    flow_target = {}
    flow_steps = {}
    total_steps = 0

    for i in empty_indices:
        current = i
        steps = 0
        current_proxy = proxy_values[current]
        while steps < max_steps:
            # Use relative tolerance: neighbor qualifies if its proxy is at least tol_fraction lower than current
            lower_neighbors = [
                j
                for j in neighbor_dict[current]
                if (current_proxy - proxy_values[j]) > tol_fraction * current_proxy
            ]
            if not lower_neighbors:
                if adaptive:
                    # Optionally, if no neighbor is found, increase tolerance (or adapt in some way)
                    tol_fraction *= adaptive_factor
                    lower_neighbors = [
                        j
                        for j in neighbor_dict[current]
                        if (current_proxy - proxy_values[j])
                        > tol_fraction * current_proxy
                    ]
                    if not lower_neighbors:
                        break
                else:
                    break
            next_current = min(lower_neighbors, key=lambda j: proxy_values[j])
            current = next_current
            current_proxy = proxy_values[current]
            steps += 1
        if steps >= min_steps:
            flow_target[i] = current
            flow_steps[i] = steps
            total_steps += steps

    average_steps = total_steps / max(len(flow_target), 1)
    logging.info(f"Average number of flow steps per tetrahedron: {average_steps:.2f}")

    # Group tetrahedra by their final sink (flow target)
    sink_groups = {}
    for i, sink in flow_target.items():
        sink_groups.setdefault(sink, []).append(i)

    final_pockets = []
    connectivity_graph = {}
    # For each sink group, build a connectivity graph among tetrahedra
    for sink, tetra_indices in sink_groups.items():
        graph = {i: [] for i in tetra_indices}
        for idx, i in enumerate(tetra_indices):
            for j in tetra_indices[idx + 1 :]:
                if len(set(simplices[i]) & set(simplices[j])) == 3:
                    graph[i].append(j)
                    graph[j].append(i)
        visited = set()
        for node in tetra_indices:
            if node not in visited:
                component = []
                stack = [node]
                while stack:
                    cur = stack.pop()
                    if cur not in visited:
                        visited.add(cur)
                        component.append(cur)
                        stack.extend(graph[cur])
                final_pockets.append(component)
                connectivity_graph[len(final_pockets) - 1] = graph

    
    final_pockets = apply_splitting_to_shared_atoms(
        final_pockets, protein_coords, flow_params.get("atom_radii")
    )
    return final_pockets, flow_steps, connectivity_graph

def _round_float_list(values, ndigits=4):
    return [round(float(v), ndigits) for v in values]


def _top_volume_signature(volumes, top_k=3, decimals=1):
    if not volumes:
        return tuple()
    top = sorted((float(v) for v in volumes), reverse=True)[:top_k]
    return tuple(round(v, decimals) for v in top)


def _signature_similarity(sig_a, sig_b):
    a = set(sig_a)
    b = set(sig_b)
    if not a and not b:
        return 1.0
    if not a or not b:
        return 0.0
    return len(a & b) / len(a | b)


def _single_alpha_run(
    protein_coords,
    simplices,
    tetra_positions,
    alpha_value,
    molecule_name=None,
    min_volume_threshold=None,
    flow_params=None,
    merge_clusters=None,
    merge_threshold=None,
):
    if min_volume_threshold is None:
        min_volume_threshold = cfg.MIN_POCKET_VOLUME
    if flow_params is None:
        flow_params = cfg.FLOW_KWARGS
    if merge_clusters is None:
        merge_clusters = cfg.MERGE_CLUSTERS
    if merge_threshold is None:
        merge_threshold = cfg.get_merge_threshold()

    alpha_mask, local_radii, local_tetra_positions, local_simplices = (
        compute_alpha_complex_from_tetrahedra(
            simplices=simplices.copy(),
            tetra_positions=tetra_positions.copy(),
            alpha_value=alpha_value,
            molecule_name=molecule_name,
            protein_coords=protein_coords,
            save_alpha_data=False,
        )
    )

    if np.sum(alpha_mask) == 0:
        return {
            "alpha": float(alpha_value),
            "num_alpha_tetrahedra": 0,
            "num_preliminary_pockets": 0,
            "num_final_pockets": 0,
            "volumes": [],
            "top_signature": tuple(),
            "mean_volume": 0.0,
            "max_volume": 0.0,
        }

    # Use the same radii that were used for alpha filtering
    # (circumradii for standard Delaunay, power radii for weighted Delaunay)
    pockets, flow_steps, connectivity_graph = flow_detection(
        simplices,
        alpha_mask,
        tetra_positions,
        protein_coords,
        flow_params,
        proxy_values=radii,  # radii is already computed earlier in this function
    )

    valid_pockets = [p for p in pockets if all(0 <= idx <= max_index for idx in p)]

    if merge_clusters:
        valid_pockets = merge_nearby_clusters(
            valid_pockets,
            local_tetra_positions,
            merge_threshold,
        )

    pocket_volumes = [
        compute_analytic_pocket_volume(local_tetra_positions, pocket)
        for pocket in valid_pockets
    ]

    filtered = [
        (pocket, volume)
        for pocket, volume in zip(valid_pockets, pocket_volumes)
        if volume >= min_volume_threshold
    ]

    if not filtered:
        return {
            "alpha": float(alpha_value),
            "num_alpha_tetrahedra": int(np.sum(alpha_mask)),
            "num_preliminary_pockets": len(valid_pockets),
            "num_final_pockets": 0,
            "volumes": [],
            "top_signature": tuple(),
            "mean_volume": 0.0,
            "max_volume": 0.0,

    filtered_volumes = list(filtered_volumes)

    return {
        "alpha": float(alpha_value),
        "num_alpha_tetrahedra": int(np.sum(alpha_mask)),
        "num_preliminary_pockets": len(valid_pockets),
        "num_final_pockets": len(filtered_volumes),

        "mean_volume": float(np.mean(filtered_volumes)),
        "max_volume": float(np.max(filtered_volumes)),
    }


def alpha_stability(
    protein_coords,
    simplices,
    tetra_positions,
    molecule_name=None,
    base_alpha=None,
    alpha_offsets=None,
    min_volume_threshold=None,
    flow_params=None,
    merge_clusters=None,
    merge_threshold=None,
):
    if base_alpha is None:
        base_alpha = cfg.get_alpha()

    if alpha_offsets is None:
        alpha_offsets = [-0.2, -0.1, 0.0, 0.1, 0.2]

    alpha_values = sorted(
        {
            round(float(base_alpha + offset), 4)
            for offset in alpha_offsets
            if (base_alpha + offset) > 0
        }
    )

    runs = []
    for alpha_value in alpha_values:
        logging.info(f"[alpha_stability] Testing alpha={alpha_value:.4f}")
        run = _single_alpha_run(
            protein_coords=protein_coords,
            simplices=simplices.copy(),
            tetra_positions=tetra_positions.copy(),
            alpha_value=alpha_value,
            molecule_name=molecule_name,
            min_volume_threshold=min_volume_threshold,
            flow_params=flow_params,
            merge_clusters=merge_clusters,
            merge_threshold=merge_threshold,
        )
        runs.append(run)

    base_run = min(runs, key=lambda r: abs(r["alpha"] - float(base_alpha)))

    pocket_counts = [r["num_final_pockets"] for r in runs]
    max_volumes = [r["max_volume"] for r in runs]
    mean_volumes = [r["mean_volume"] for r in runs]

    count_range = int(max(pocket_counts) - min(pocket_counts)) if pocket_counts else 0
    max_volume_cv = (
        float(np.std(max_volumes) / np.mean(max_volumes))
        if max_volumes and np.mean(max_volumes) > 0
        else 0.0
    )
    mean_volume_cv = (
        float(np.std(mean_volumes) / np.mean(mean_volumes))
        if mean_volumes and np.mean(mean_volumes) > 0
        else 0.0
    )

    signature_similarities = [
        _signature_similarity(base_run["top_signature"], r["top_signature"])
        for r in runs
    ]
    avg_signature_similarity = (
        float(np.mean(signature_similarities)) if signature_similarities else 0.0
    )

    stable = (
        count_range <= 1
        and max_volume_cv <= 0.25
        and avg_signature_similarity >= 0.6
    )

    summary = {
        "base_alpha": float(base_alpha),
        "tested_alphas": [float(r["alpha"]) for r in runs],
        "runs": [
            {
                "alpha": r["alpha"],
                "num_alpha_tetrahedra": r["num_alpha_tetrahedra"],
                "num_preliminary_pockets": r["num_preliminary_pockets"],
                "num_final_pockets": r["num_final_pockets"],
                "volumes": r["volumes"],
                "top_signature": list(r["top_signature"]),
                "mean_volume": round(float(r["mean_volume"]), 4),
                "max_volume": round(float(r["max_volume"]), 4),
            }
            for r in runs
        ],
        "metrics": {
            "count_range": count_range,
            "max_volume_cv": round(max_volume_cv, 4),
            "mean_volume_cv": round(mean_volume_cv, 4),
            "avg_top_signature_similarity": round(avg_signature_similarity, 4),
        },
        "stable": bool(stable),
    }

    logging.info(
        "[alpha_stability] stable=%s | count_range=%s | max_volume_cv=%.4f | avg_top_signature_similarity=%.4f",
        summary["stable"],
        summary["metrics"]["count_range"],
        summary["metrics"]["max_volume_cv"],
        summary["metrics"]["avg_top_signature_similarity"],
    )

    return summary

def cgal_weighted_delaunay(protein_coords, radii):
    if USE_CGAL:
        try:
            from pocketlab import build_regular_triangulation

            result = build_regular_triangulation(protein_coords, radii)

            if isinstance(result, dict) and result.get("ok") and "cells_idx" in result:
                simplices = np.asarray(result["cells_idx"], dtype=int)

                    f"[CGAL] Weighted Delaunay built successfully: {len(simplices)} cells."
                )
                return simplices, result

            logging.warning(

            )

                f"[CGAL] pocketlab unavailable or failed ({e}). Falling back to SciPy Delaunay."
            )

    simplices = Delaunay(protein_coords).simplices
    logging.info(f"[SciPy] Standard Delaunay built: {len(simplices)} simplices.")
    return simplices, None

#!/usr/bin/env python
# -*- coding: utf-8 -*-
"""
dual_sets.py

Module for computing the dual set of a pocket and for extracting dual mouth components.

The idea is as follows:
  1. Compute the closure of the pocket, i.e. collect all faces (triangles) from every tetrahedron
     in the pocket.
  2. For each face, assign a "filtration value" (we use the minimum circumsphere radius of any
     tetrahedron that contributed that face).
  3. Compute the dual set by keeping only those faces whose filtration value is at least a given
     threshold (usually the chosen alpha value).
  4. Compute the boundary of the dual set (those faces that appear only once) and then extract the
     connected components, each of which corresponds to a pocket mouth.
     
Note: This is a simplified version based on the pseudo‐algorithm. You may need to further refine it
for your specific application.
"""

import numpy as np
import logging
from collections import defaultdict
from scipy.spatial import ConvexHull


def get_faces_from_tetrahedron(tet, decimals=3):
    """
    Given a tetrahedron (4x3 array of coordinates), return a list of its 4 triangle faces.
    Each face is represented as a tuple of 3 vertex tuples, with coordinates rounded to a
    specified number of decimals (to allow consistent comparison).
    """
    # Round vertices
    vertices = [tuple(np.round(v, decimals=decimals)) for v in tet]
    # Tetrahedron faces: there are 4 triangles
    face_indices = [(0, 1, 2), (0, 1, 3), (0, 2, 3), (1, 2, 3)]
    faces = [tuple(sorted([vertices[i] for i in inds])) for inds in face_indices]
    return faces


def compute_closure_of_pocket(pocket_indices, tetra_positions, decimals=3):
    """
    Compute the closure of a pocket defined by a list of tetrahedron indices.
    The closure is the union of all faces (triangles) of the tetrahedra.

    Returns:
      closure: a dictionary mapping each face (tuple) to its occurrence count.
    """
    closure = {}
    for idx in pocket_indices:
        tet = tetra_positions[idx]
        faces = get_faces_from_tetrahedron(tet, decimals=decimals)
        for face in faces:
            closure[face] = closure.get(face, 0) + 1
    return closure


def compute_boundary_from_closure(closure):
    """
    The boundary of the closure is defined as the set of faces that appear only once.

    Returns:
      boundary: a set of faces.
    """
    boundary = {face for face, count in closure.items() if count == 1}
    return boundary


def compute_face_filtration_values(pocket_indices, tetra_positions, radii, decimals=3):
    """
    For each face in the pocket, compute its filtration value as the minimum circumsphere radius
    among the tetrahedra (in the pocket) that contributed that face.

    Parameters:
      pocket_indices: list of indices of tetrahedra in the pocket.
      tetra_positions: array of shape (n_tetra, 4, 3) with tetrahedron coordinates.
      radii: array of circumsphere radii corresponding to tetra_positions.

    Returns:
      face_filtration: dict mapping face (tuple) to a filtration value.
    """
    face_filtration = {}
    for idx in pocket_indices:
        tet = tetra_positions[idx]
        face_value = radii[
            idx
        ]  # Use the tetrahedron’s circumsphere radius as the proxy filtration value
        faces = get_faces_from_tetrahedron(tet, decimals=decimals)
        for face in faces:
            if face in face_filtration:
                face_filtration[face] = min(face_filtration[face], face_value)
            else:
                face_filtration[face] = face_value
    return face_filtration


def compute_dual_set_from_closure(closure, face_filtration, alpha_value):
    """
    Compute the dual set of a pocket by removing from the closure all faces whose filtration value is
    less than the given alpha_value.

    Returns:
      dual_set: set of faces (triangles) that form the dual representation.
    """
    dual_set = {
        face
        for face in closure
        if face_filtration.get(face, float("inf")) >= alpha_value
    }
    return dual_set


def build_connectivity_graph(simplices):
    """
    Build a connectivity graph among simplices.
    Two simplices are considered connected if they share at least two vertices (i.e. an edge).

    Parameters:
      simplices: iterable of simplices (e.g., faces represented as tuples of vertex coordinates).

    Returns:
      graph: dict mapping each simplex to a list of neighboring simplices.
    """
    graph = defaultdict(list)
    simplices = list(simplices)
    for i, s in enumerate(simplices):
        for j in range(i + 1, len(simplices)):
            t = simplices[j]
            # For triangles, sharing two vertices means they share an edge.
            if len(set(s) & set(t)) >= 2:
                graph[s].append(t)
                graph[t].append(s)
    return graph


def find_connected_components(graph):
    """
    Find connected components in a graph.

    Parameters:
      graph: dict mapping node to list of neighbors.

    Returns:
      components: list of sets, each set representing a connected component.
    """
    visited = set()
    components = []
    for node in graph:
        if node not in visited:
            component = set()
            stack = [node]
            while stack:
                cur = stack.pop()
                if cur not in visited:
                    visited.add(cur)
                    component.add(cur)
                    stack.extend(graph[cur])
            components.append(component)
    return components


def compute_dual_mouths(boundary_faces, global_dual=None):
    """
    Given the boundary faces of a pocket (from the closure) and an optional global dual complex,
    compute the connected components that represent the mouth(s) of the pocket.

    Parameters:
      boundary_faces: set of faces (triangles) forming the boundary of the pocket.
      global_dual: set of faces belonging to the dual complex of the entire structure (optional).

    Returns:
      mouth_components: list of sets, each set representing one connected mouth.
    """
    # If provided, remove faces that are in the global dual complex.
    if global_dual is not None:
        filtered_boundary = {face for face in boundary_faces if face not in global_dual}
    else:
        filtered_boundary = boundary_faces

    # Build connectivity graph among the remaining boundary faces.
    graph = build_connectivity_graph(filtered_boundary)
    # Extract connected components.
    mouth_components = find_connected_components(graph)
    return mouth_components

def compute_dual_set_for_pocket(
    pocket_indices, tetra_positions, radii, alpha_value, decimals=3
):
    """
    Given a pocket (indices into tetra_positions), compute:
      - the closure (all faces),
      - the filtration value for each face,
      - the dual set (faces with filtration value >= alpha_value),
      - and the boundary of the dual set.

    Returns a dictionary with keys:
      'closure', 'face_filtration', 'dual_set', and 'boundary'
    """
    closure = compute_closure_of_pocket(
        pocket_indices, tetra_positions, decimals=decimals
    )
    face_filtration = compute_face_filtration_values(
        pocket_indices, tetra_positions, radii, decimals=decimals
    )
    dual_set = compute_dual_set_from_closure(closure, face_filtration, alpha_value)
    boundary = compute_boundary_from_closure(closure)
    return {
        "closure": closure,
        "face_filtration": face_filtration,
        "dual_set": dual_set,
        "boundary": boundary,
    }

from pathlib import Path
import os

ROOT = Path(__file__).resolve().parent
OUTFILE = ROOT / "project_dump.txt"

EXCLUDE_DIRS = {
    ".git", ".venv", "venv", "__pycache__",
    ".pytest_cache", ".mypy_cache",
    "build", "dist", "results", "result",
    "output", "data", ".idea", ".vscode",
}

EXCLUDE_FILES = {
    ".DS_Store",
}

TEXT_EXTS = {
    ".py", ".txt", ".md", ".toml", ".cfg", ".ini", ".yaml", ".yml",
    ".json", ".js", ".ts", ".sh", ".ps1", ".bat", ".cmd", ".rst",
}

def is_text_file(path: Path) -> bool:
    return path.suffix.lower() in TEXT_EXTS

def should_skip_dir(name: str) -> bool:
    return name in EXCLUDE_DIRS

def should_skip_file(name: str) -> bool:
    return name in EXCLUDE_FILES or name.endswith((".pyc", ".pyo", ".pyd"))

def build_tree(root: Path) -> str:
    lines = [root.name]

    def walk(dir_path: Path, prefix: str = ""):
        items = []
        for p in sorted(dir_path.iterdir(), key=lambda x: (not x.is_dir(), x.name.lower())):
            if p.is_dir() and should_skip_dir(p.name):
                continue
            if p.is_file() and should_skip_file(p.name):
                continue
            items.append(p)

        for i, p in enumerate(items):
            last = i == len(items) - 1
            branch = "└── " if last else "├── "
            lines.append(prefix + branch + p.name)
            if p.is_dir():
                walk(p, prefix + ("    " if last else "│   "))

    walk(root)
    return "\n".join(lines)

def collect_file_contents(root: Path) -> str:
    parts = []
    for dirpath, dirs, files in os.walk(root):
        dirs[:] = [d for d in dirs if not should_skip_dir(d)]
        for filename in sorted(files):
            if should_skip_file(filename):
                continue
            path = Path(dirpath) / filename
            if path.resolve() == OUTFILE.resolve():
                continue
            if not is_text_file(path):
                continue
            try:
                content = path.read_text(encoding="utf-8", errors="replace")
            except Exception as e:
                content = f"[ERROR READING FILE: {e}]"
            parts.append(f"### FILE: {path.relative_to(root)}\n{content}\n")
    return "\n---\n\n".join(parts)

def main():
    tree = build_tree(ROOT)
    contents = collect_file_contents(ROOT)
    OUTFILE.write_text(
        f"ROOT: {ROOT}\n\nPROJECT TREE\n{tree}\n\n{'='*80}\n\nFILE CONTENTS\n{'='*80}\n\n{contents}\n",
        encoding="utf-8"
    )
    print(f"Written: {OUTFILE}")

if __name__ == "__main__":
    main()

import numpy as np

from ..pocket_detection import compute_analytic_pocket_volume


def compute_pocket_depth_from_tetra(tetra_positions, pocket):
    if not pocket:
        return 0.0

    pocket_points = tetra_positions[pocket].reshape(-1, 3)
    centroid = np.mean(pocket_points, axis=0)
    distances = np.linalg.norm(pocket_points - centroid, axis=1)
    return float(np.max(distances))


def compute_geometry_features(tetra_positions, pocket):
    volume = compute_analytic_pocket_volume(tetra_positions, pocket)
    depth = compute_pocket_depth_from_tetra(tetra_positions, pocket)
    n_tetra = len(pocket)

    return {
        "volume": float(volume),
        "depth": float(depth),
        "n_tetra": int(n_tetra),
    }

import numpy as np
from scipy.spatial import KDTree


def fibonacci_sphere(n_points):
    points = []
    phi = np.pi * (3.0 - np.sqrt(5.0))
    for i in range(n_points):
        y = 1.0 - (i / float(max(n_points - 1, 1))) * 2.0
        radius = np.sqrt(max(0.0, 1.0 - y * y))
        theta = phi * i
        x = np.cos(theta) * radius
        z = np.sin(theta) * radius
        points.append([x, y, z])
    return np.asarray(points, dtype=float)


def ray_sphere_intersection(origin, direction, center, radius):
    oc = origin - center
    b = 2.0 * np.dot(oc, direction)
    c = np.dot(oc, oc) - radius * radius
    disc = b * b - 4.0 * c

    if disc < 0:
        return None

    sqrt_disc = np.sqrt(disc)
    t1 = (-b - sqrt_disc) / 2.0
    t2 = (-b + sqrt_disc) / 2.0
    valid_ts = [t for t in (t1, t2) if t > 1e-6]
    return min(valid_ts) if valid_ts else None


def _normalize_01(value, low, high):
    if high <= low:
        return 0.0
    return float(np.clip((value - low) / (high - low), 0.0, 1.0))


def compute_hybrid_occlusion_features(
    candidate,
    protein_coords,
    protein_atoms,
    atom_radii,
    sasa_by_residue=None,
    num_rays=96,
    max_distance=12.0,
    neighbor_radius=16.0,
    low_sasa_threshold=5.0,
    kd_tree=None,
):
    center = np.asarray(candidate.representative_point, dtype=float)
    protein_coords = np.asarray(protein_coords, dtype=float)
    atom_radii = np.asarray(atom_radii, dtype=float)

    if kd_tree is None:
        kd_tree = KDTree(protein_coords)

    neighbor_indices = kd_tree.query_ball_point(center, r=neighbor_radius)

    if not neighbor_indices:
        return {
            "ray_occlusion_fraction": 0.0,
            "ray_escape_fraction": 1.0,
            "ray_mean_hit_distance": float(max_distance),
            "ray_min_hit_distance": float(max_distance),
            "hit_residue_sasa_mean": 0.0,
            "low_sasa_hit_fraction": 0.0,
            "hybrid_occlusion_score": 0.0,
            "occlusion_debug": {
                "num_hits": 0,
                "num_rays": int(num_rays),
                "hit_atom_indices": [],
            },
        }

    local_coords = protein_coords[neighbor_indices]
    local_radii = atom_radii[neighbor_indices]
    directions = fibonacci_sphere(num_rays)

    hit_distances = []
    hit_atom_indices = []
    hit_residues = []

    for direction in directions:
        direction = direction / np.linalg.norm(direction)
        best_t = None
        best_atom_index = None

        for local_i, (atom_center, atom_radius) in enumerate(zip(local_coords, local_radii)):
            t_hit = ray_sphere_intersection(center, direction, atom_center, atom_radius)
            if t_hit is None or t_hit > max_distance:
                continue
            if best_t is None or t_hit < best_t:
                best_t = t_hit
                best_atom_index = neighbor_indices[local_i]

        if best_atom_index is not None:
            hit_distances.append(best_t)
            hit_atom_indices.append(int(best_atom_index))
            residue_id = str(protein_atoms[best_atom_index].get("resid", ""))
            hit_residues.append(residue_id)

    num_hits = len(hit_distances)
    ray_occlusion_fraction = num_hits / float(num_rays)
    ray_escape_fraction = 1.0 - ray_occlusion_fraction
    ray_mean_hit_distance = float(np.mean(hit_distances)) if hit_distances else float(max_distance)
    ray_min_hit_distance = float(np.min(hit_distances)) if hit_distances else float(max_distance)

    sasa_values = []

    hit_residue_sasa_mean = float(np.mean(sasa_values)) if sasa_values else 0.0
    low_sasa_hit_fraction = (
        float(np.mean([value <= low_sasa_threshold for value in sasa_values]))
        if sasa_values else 0.0
    )

    mean_hit_distance_norm = _normalize_01(ray_mean_hit_distance, 0.0, max_distance)
    sasa_norm = _normalize_01(hit_residue_sasa_mean, 0.0, 30.0)

    hybrid_occlusion_score = (
        0.55 * ray_occlusion_fraction +
        0.25 * (1.0 - mean_hit_distance_norm) +
        0.20 * (1.0 - sasa_norm)
    )

    return {
        "ray_occlusion_fraction": float(ray_occlusion_fraction),
        "ray_escape_fraction": float(ray_escape_fraction),
        "ray_mean_hit_distance": float(ray_mean_hit_distance),
        "ray_min_hit_distance": float(ray_min_hit_distance),
        "hit_residue_sasa_mean": float(hit_residue_sasa_mean),
        "low_sasa_hit_fraction": float(low_sasa_hit_fraction),
        "hybrid_occlusion_score": float(hybrid_occlusion_score),
        "occlusion_debug": {
            "num_hits": int(num_hits),
            "num_rays": int(num_rays),
            "hit_atom_indices": hit_atom_indices[:128],
        },
    }

    num_pockets = len(result.get("ranked_pockets", []))
    if target_min <= num_pockets <= target_max:
        return 0
    else:
        penalty = abs(
            num_pockets - (target_min if num_pockets < target_min else target_max)
        )
        return penalty

    alpha_val, vol_thresh, merge_thresh, flow_tol, flow_max, flow_adapt = combo

            "score": score,

        best_score = score
        best_params = combo

    FLOW_KWARGS["tol_fraction"] = old_tol
    FLOW_KWARGS["max_steps"] = old_max_steps
    FLOW_KWARGS["adaptive_factor"] = old_adaptive_factor

Calcolo di feature di occlusione basate su raycasting e SASA.

import numpy as np
from scipy.spatial import KDTree


def fibonacci_sphere(n_points):
    """
    Genera n_points direzioni uniformemente distribuite su una sfera
    usando il campionamento di Fibonacci.
    """
    points = []
    phi = np.pi * (3.0 - np.sqrt(5.0))
    for i in range(n_points):
        y = 1.0 - (i / float(max(n_points - 1, 1))) * 2.0
        radius = np.sqrt(max(0.0, 1.0 - y * y))
        theta = phi * i
        x = np.cos(theta) * radius
        z = np.sin(theta) * radius
        points.append([x, y, z])
    return np.asarray(points, dtype=float)


def ray_sphere_intersection(origin, direction, center, radius):
    """
    Calcola l'intersezione di un raggio (origine + direzione) con una sfera.
    Restituisce la distanza t del primo hit (t > 0), oppure None se non c'è··intersezione.
    """
    oc = origin - center
    b = 2.0 * np.dot(oc, direction)
    c = np.dot(oc, oc) - radius * radius
    disc = b * b - 4.0 * c

    if disc < 0:
        return None

    sqrt_disc = np.sqrt(disc)
    t1 = (-b - sqrt_disc) / 2.0
    t2 = (-b + sqrt_disc) / 2.0
    valid_ts = [t for t in (t1, t2) if t > 1e-6]
    return min(valid_ts) if valid_ts else None


def _normalize_01(value, low, high):
    """Normalizza un valore in [0, 1] dato un intervallo [low, high]."""
    if high <= low:
        return 0.0
    return float(np.clip((value - low) / (high - low), 0.0, 1.0))


def compute_occlusion_features(

    local_radii = atom_radii[neighbor_indices]
    directions = fibonacci_sphere(num_rays)

    hit_distances = []
    hit_atom_indices = []
    hit_residues = []

    for direction in directions:

            t_hit = ray_sphere_intersection(center, direction, atom_center, atom_radius)
            if t_hit is None or t_hit > max_distance:
                continue

                best_atom_index = neighbor_indices[local_i]

        if best_atom_index is not None:
            hit_distances.append(best_t)
            hit_atom_indices.append(int(best_atom_index))
            residue_id = str(protein_atoms[best_atom_index].get("resid", ""))

    num_hits = len(hit_distances)
    ray_occlusion_fraction = num_hits / float(num_rays)

    ray_mean_hit_distance = float(np.mean(hit_distances)) if hit_distances else float(max_distance)
    ray_min_hit_distance = float(np.min(hit_distances)) if hit_distances else float(max_distance)

    # SASA dei residui colpiti
    sasa_values = []
    if sasa_by_residue:
        for residue_id in set(hit_residues):
            sasa_values.append(float(sasa_by_residue.get(str(residue_id), 0.0)))

    hit_residue_sasa_mean = float(np.mean(sasa_values)) if sasa_values else 0.0

    sasa_norm = _normalize_01(hit_residue_sasa_mean, 0.0, 30.0)

    # Score ibrido: pesi empirici, da ottimizzare
    hybrid_occlusion_score = (
        0.55 * ray_occlusion_fraction +
        0.25 * (1.0 - mean_hit_distance_norm) +
        0.20 * (1.0 - sasa_norm)
    )

    return {
        "ray_occlusion_fraction": float(ray_occlusion_fraction),
        "ray_escape_fraction": float(ray_escape_fraction),

def build_pocket_candidates(ranked_pockets, tetra_positions):
    candidates = []

    for idx, pocket in enumerate(ranked_pockets, start=1):
        pocket_points = tetra_positions[pocket].reshape(-1, 3)
        representative_point = np.mean(pocket_points, axis=0).tolist()

        candidates.append(
            PocketCandidate(
                pocket_id=idx,
                tetra_indices=list(pocket),
                representative_point=representative_point,
            )
        )

    return candidates


def enrich_with_geometry(candidates, tetra_positions):

        feats = compute_geometry_features(tetra_positions, candidate.tetra_indices)
        candidate.volume = feats["volume"]
        candidate.depth = feats["depth"]
        candidate.features.update(feats)
    return candidates


def enrich_with_mouth(candidates, ranked_pockets, simplices, protein_coords, ligand_coords=None):

        ranked_pockets=ranked_pockets,

        protein_coords=protein_coords,

# Analytic pocket volume computation defined here
def compute_analytic_pocket_volume(tetra_positions, pocket_indices):
    pocket_tetra = tetra_positions[pocket_indices]
    vec1 = pocket_tetra[:, 1] - pocket_tetra[:, 0]
    vec2 = pocket_tetra[:, 2] - pocket_tetra[:, 0]
    vec3 = pocket_tetra[:, 3] - pocket_tetra[:, 0]
    volumes = np.abs(np.einsum("ij,ij->i", np.cross(vec1, vec2), vec3)) / 6.0
    return float(np.sum(volumes))


def detect_pockets(
    protein_coords,
    simplices=None,
    tetra_positions=None,
    alpha_mask=None,
    min_volume_threshold=None,
    flow_params=None,

):
    if min_volume_threshold is None:

    if flow_params is None:
        flow_params = cfg.FLOW_KWARGS

    if merge_clusters is None:
        merge_clusters = cfg.MERGE_CLUSTERS

    if merge_threshold is None:
        merge_threshold = cfg.get_merge_threshold()

    logging.info(f"CONFIG Merge threshold currently set to {merge_threshold}")

    if tetra_positions is None:
        tetra_positions = protein_coords[simplices]

    logging.info(f"Delaunay triangulation: {len(simplices)} tetrahedra found.")
    max_index = len(protein_coords) - 1

    if np.max(simplices) > max_index:
        raise ValueError(
            f"ERROR: Some simplex indices exceed protein_coords size ({max_index})."
        )

    alpha_value = cfg.get_alpha()

    if alpha_mask is None or radii is None:
        alpha_mask, radii, tetra_positions, simplices = compute_alpha_complex_from_tetrahedra(
            simplices,
            tetra_positions,
            alpha_value,
            molecule_name,
            protein_coords,
            radii=radii,
            filter_by_sasa=filter_by_sasa,
            sasa_threshold=sasa_thresold,

        )

    logging.info(f"Using alpha shape with alpha {alpha_value}")

    )
    if np.sum(alpha_mask) == 0:
        logging.warning("Alpha shape filtered out all tetrahedra. No pockets detected.")
        return [], [], []
    pockets, flow_steps, connectivity_graph = flow_detection(
        simplices, alpha_mask, tetra_positions, protein_coords, flow_params,proxy_values=radii,
    )
    logging.info(f"Flow detection found {len(pockets)} preliminary pockets.")

        if all(0 <= idx <= max_index for idx in pocket):
            valid_pockets.append(pocket)
    if merge_clusters:

        )
    else:
        pockets = valid_pockets
    pocket_volumes = [

    ]

            pocket, tetra_positions, radii, alpha_value=alpha_value, decimals=3

        "dual_sets_info": dual_sets_info,

    }

# ranking.py
import numpy as np
import logging
from .config import RANKING_METHOD


def compute_ranking_scores(volumes, pockets, flowsteps, connectivitygraph, tetrapositions):
    """Dispatcher: sceglie il metodo in base a config.RANKING_METHOD."""
    method = RANKING_METHOD
    logging.info(f"[ranking] Using method: {method}")
    if method == "percentile_log":
        return _ranking_percentile_log(volumes, pockets, flowsteps, connectivitygraph)
    else:
        return _ranking_baseline(volumes, pockets, flowsteps, connectivitygraph)


# ── Baseline (metodo originale, invariato) ───────────────────────────────────
def _ranking_baseline(volumes, pockets, flowsteps, connectivitygraph):
    scores = []
    for i, (volume, pocket) in enumerate(zip(volumes, pockets)):
        nsteps       = flowsteps.get(i, 1)
        nconnections = len(connectivitygraph.get(i, []))
        norm_volume      = volume
        norm_steps       = np.log1p(nsteps)
        norm_connectivity = np.log1p(nconnections)
        score = 1.0 * norm_volume + 0.5 * norm_steps + 0.2 * norm_connectivity
        logging.debug(f"Pocket {i+1} volume={volume:.2f}, steps={nsteps}, "
                      f"connections={nconnections}, score={score:.2f}")
        scores.append(score)
    return scores


# ── Variante: percentile-log normalizzata ────────────────────────────────────
def _ranking_percentile_log(volumes, pockets, flowsteps, connectivitygraph):
    """
    Normalizza ogni componente sul range osservato (percentile 5-95),
    poi combina con pesi identici alla baseline ma su scala [0,1].
    Più stabile su distribuzioni skewed di volumi.
    """
    if not volumes:
        return []

    vols  = np.array(volumes, dtype=float)
    steps = np.array([flowsteps.get(i, 1) for i in range(len(pockets))], dtype=float)
    conns = np.array([len(connectivitygraph.get(i, [])) for i in range(len(pockets))], dtype=float)

    def _pct_norm(arr):
        lo, hi = np.percentile(arr, 5), np.percentile(arr, 95)
        if hi == lo:
            return np.ones_like(arr) * 0.5
        return np.clip((arr - lo) / (hi - lo), 0.0, 1.0)

    norm_v = _pct_norm(np.log1p(vols))
    norm_s = _pct_norm(np.log1p(steps))
    norm_c = _pct_norm(np.log1p(conns))

    scores = (1.0 * norm_v + 0.5 * norm_s + 0.2 * norm_c).tolist()

    for i, score in enumerate(scores):
        logging.debug(f"[v2] Pocket {i+1} score={score:.4f}")

    return scores

def compute_pocket_distances(pocket_points, ligand_coords):
    if ligand_coords.size == 0 or pocket_points.size == 0:
        logging.warning("Ligand or pocket points are missing!")
        return {"min": None, "max": None, "avg": None}

    distances = cdist(pocket_points, ligand_coords)
    return {
        "min": np.min(distances),
        "max": np.max(distances),
        "avg": np.mean(distances),
    }


class DummyAtom:
    def __init__(self, coord):
        self.coord = np.array(coord, dtype=float)

        return self.coord


def match_atoms_by_residue_and_name(
    fixed_atoms,
    moving_atoms,
    allowed_atom_names=["CA"],
):
    fixed_filtered = [a for a in fixed_atoms if a["atom_name"] in allowed_atom_names]
    moving_filtered = [a for a in moving_atoms if a["atom_name"] in allowed_atom_names]

    fixed_dict = {(a["resid"], a["atom_name"]): a["coord"] for a in fixed_filtered}
    moving_dict = {(a["resid"], a["atom_name"]): a["coord"] for a in moving_filtered}

    common_keys = sorted(set(fixed_dict.keys()) & set(moving_dict.keys()))
    if len(common_keys) == 0:
        logging.warning("No matching atoms found between fixed and moving sets.")
        return [], []

    fixed_coords = [fixed_dict[k] for k in common_keys]
    moving_coords = [moving_dict[k] for k in common_keys]

    logging.info(
        f"Matched {len(common_keys)} atoms for alignment based on {allowed_atom_names}."
    )
    return fixed_coords, moving_coords


def make_dummy_atom(coord):
    return Atom(
        name="X",
        coord=coord,
        bfactor=0.0,
        occupancy=1.0,
        altloc=" ",
        fullname=" X ",
        serial_number=1,
        element="X",
    )


def align_using_superimposer(fixed_coords, moving_coords, return_transform=False):
    fixed_coords = np.array(fixed_coords)
    moving_coords = np.array(moving_coords)

    n = min(len(fixed_coords), len(moving_coords))
    if len(fixed_coords) != len(moving_coords):
        logging.warning(
            f"Different number of atoms: fixed {len(fixed_coords)} vs moving {len(moving_coords)}. Using the first {n} atoms."
        )
        fixed_coords = fixed_coords[:n]
        moving_coords = moving_coords[:n]

    fixed_atoms = [make_dummy_atom(coord) for coord in fixed_coords]
    moving_atoms = [make_dummy_atom(coord) for coord in moving_coords]

    sup = Superimposer()
    sup.set_atoms(fixed_atoms, moving_atoms)

    R, t = sup.rotran
    aligned = np.dot(moving_coords, R) + t
    rmsd = sup.rms

    logging.info(f"Alignment RMSD: {rmsd:.3f} Å")

    if return_transform:
        return aligned, rmsd, (R, t)
    return aligned, rmsd


def analyze_top_pockets(result):
    """
    Determines the first validated pocket based on the chosen validation method.
    Returns a tuple: (top_index, validation_tags)
    where top_index is the 1-based index of the first pocket that validated (or None),
    and validation_tags is a list (one per pocket) indicating the chosen method if validated, or "None".
    This function checks the unified field 'ligand_validation'.
    """
    method = cfg.VALIDATION_METHOD.lower()
    validations = result.get("ligand_validation", [])

    top_index = None
    validation_tags = []

    for idx, valid in enumerate(validations):
        if valid:
            validation_tags.append(method.capitalize())
            if top_index is None:
                top_index = idx + 1
        else:
            validation_tags.append("None")

    return top_index, validation_tags

    result["validation_methods"] = validation_methods
    result["ligand_validation"] = ligand_validation
    result["ligand_containment_mesh"] = ligand_containment_mesh

    )

    # Save pocket PDBs
    protein_df = ppdb.df["ATOM"]

            np.concatenate([tetra_positions[i] for i in pocket], axis=0),
            axis=0,
        )
        _, idx = tree.query(pocket_atoms, k=1)
        pocket_df = protein_df.iloc[idx].copy()
        pocket_df["b_factor"] = rank
        pdb_out = PandasPdb()
        pdb_out.df["ATOM"] = pocket_df
        pdb_out.to_pdb(os.path.join(output_pdb_dir, f"{base}_ranked_{rank}.pdb"))
        logging.info(f"Pocket {rank} saved.")

    save_pocket_properties(result, base, cfg.OUTPUT_DIR)
    logging.info(
        f"=== Done: {pdb_path} | pockets={len(ranked_pockets)}, top={result['step_to_ligand']} ==="
    )
    return convert_numpy_types(result)

def run_analysis_mode(pdb_sources, analysis_type="single", correspondence_df=None):
    results = []
    out_dir = os.path.join(cfg.OUTPUT_DIR, cfg.VERSION_TAG)
    os.makedirs(out_dir, exist_ok=True)

    if analysis_type == "paired" and correspondence_df is not None:
        for _, row in correspondence_df.iterrows():
            b_name = row["bounded"]
            u_name = row["unbounded"]
            b_path = os.path.join(cfg.BOUNDED_DIR, f"{b_name}.pdb")
            u_path = os.path.join(cfg.UNBOUNDED_DIR, f"{u_name}.pdb")

            if not os.path.exists(b_path) or not os.path.exists(u_path):
                logging.warning(f"Skipping pair {b_name}/{u_name}: missing file(s).")
                continue

            b_res = process_pdb(b_path)
            if not b_res or not b_res.get("ranked_pockets"):
                logging.warning(f"Skipping {b_name}: no pockets.")
                continue

            u_res = process_pdb(u_path)
            if not u_res or not u_res.get("ranked_pockets"):
                logging.warning(f"Skipping {u_name}: no pockets.")
                continue

            ALIGN_ATOMS = ["CA"]
            fixed_c, moving_c = match_atoms_by_residue_and_name(
                b_res["protein_atoms"],
                u_res["protein_atoms"],
                allowed_atom_names=ALIGN_ATOMS,
            )
            _, rmsd, (R, t) = align_using_superimposer(
                fixed_c,
                moving_c,
                return_transform=True,
            )
            u_res["protein_coords"] = (
                np.dot(np.array(u_res["protein_coords"]), R) + t
            ).tolist()

import math


def compute_basic_score(candidate):
    volume = float(candidate.features.get("volume", 0.0))
    depth = float(candidate.features.get("depth", 0.0))
    mouth_area = float(candidate.features.get("mouth_area", 0.0) or 0.0)
    rim_atom_count = float(candidate.features.get("rim_atom_count", 0.0))

    score = (
        1.0 * volume
        + 0.35 * depth
        + 0.10 * math.log1p(mouth_area)
        + 0.05 * math.log1p(rim_atom_count)
    )
    return float(score)

"""

import os

from ..config import VALIDATION_METHOD


def np_converter(obj):
    if isinstance(obj, np.ndarray):
        return obj.tolist()
    elif isinstance(obj, (np.integer, np.floating)):
        return obj.item()
    elif isinstance(obj, list):
        return [np_converter(item) for item in obj]
    elif isinstance(obj, dict):
        return {key: np_converter(value) for key, value in obj.items()}
    return obj


def convert_numpy_types(obj):
    if isinstance(obj, np.ndarray):
        return obj.tolist()
    elif isinstance(obj, np.integer):
        return int(obj)
    elif isinstance(obj, np.floating):
        return float(obj)
    elif isinstance(obj, (np.bool_,)):
        return bool(obj)
    elif isinstance(obj, tuple):
        return [convert_numpy_types(x) for x in obj]
    elif isinstance(obj, set):
        return [convert_numpy_types(x) for x in obj]
    elif isinstance(obj, list):
        return [convert_numpy_types(x) for x in obj]
    elif isinstance(obj, dict):
        new_dict = {}
        for k, v in obj.items():
            # Ensure keys are acceptable
            if not isinstance(k, (str, int, float, bool)) and k is not None:
                k = str(k)
            new_dict[k] = convert_numpy_types(v)
        return new_dict
    return obj


def save_json(data, filename):
    with open(filename, "w") as f:
        json.dump(convert_numpy_types(data), f, indent=4)


def load_json(in_path):
    if not os.path.exists(in_path):
        logging.warning(f"JSON file not found: {in_path}")
        return {}
    with open(in_path, "r") as f:
        content = f.read().strip()
        if not content:
            return {}
        try:
            return json.loads(content)
        except json.decoder.JSONDecodeError as e:
            logging.warning(f"JSON decode error in {in_path}: {e}")
            return {}

"""
geometry_utils.py

Module containing functions for:
- Computing circumsphere radii of tetrahedra.
- Estimating pocket volume using Monte Carlo integration.
- Validating pockets via mesh extrusion and fake sphere tests.
"""

import os
import numpy as np
from scipy.spatial import Delaunay, ConvexHull
from scipy.spatial.distance import cdist
from sklearn.decomposition import PCA
import trimesh
from ..config import (
    SAVE_EXTRUDED_MESHES,
    OUTPUT_DIR,
    MESH_EXTRUSION_DISTANCE,
    LIGAND_THRESHOLD,
)


def get_unique_vertices(tetra_positions, pocket_indices):
    vertices = []
    for idx in pocket_indices:
        vertices.extend(tetra_positions[idx])
    vertices = np.array(vertices)
    return np.unique(vertices, axis=0)


def get_convex_hull_points(points):
    if len(points) < 4:
        return points
    hull = ConvexHull(points)
    return points[hull.vertices]


def validate_ligand_in_extruded_mesh(
    mouth_coords, ligand_coords, extrusion_distance=MESH_EXTRUSION_DISTANCE
):
    if len(mouth_coords) < 3 or ligand_coords.size == 0:
        return False, None, None

    from sklearn.decomposition import PCA
    from scipy.spatial import ConvexHull
    import logging
    import trimesh

    try:
        # Flatten the mouth coordinates using PCA
        pca = PCA(n_components=2)
        flat_2d = pca.fit_transform(mouth_coords)

        # Create 2D convex hull of the flattened points
        hull_2d = ConvexHull(flat_2d)
        ordered_indices = hull_2d.vertices
        cap_3d = mouth_coords[ordered_indices]

        # Compute extrusion normal
        normal_3d = np.cross(cap_3d[1] - cap_3d[0], cap_3d[2] - cap_3d[0])
        if np.linalg.norm(normal_3d) == 0:
            return False, None, None
        normal_3d /= np.linalg.norm(normal_3d)

        # Extrude cap
        extruded = cap_3d + extrusion_distance * normal_3d
        all_vertices = np.vstack([cap_3d, extruded])
        num = len(cap_3d)

        # Construct mesh faces
        faces = []
        for i in range(1, num - 1):
            faces.append([0, i, i + 1])
        for i in range(1, num - 1):
            faces.append([num, num + i + 1, num + i])
        for i in range(num):
            j = (i + 1) % num
            faces.append([i, j, j + num])
            faces.append([i, j + num, i + num])

        # Build the mesh
        mesh = trimesh.Trimesh(vertices=all_vertices, faces=faces, process=True)

        # Check if ligand atoms are inside the mesh
        inside_mask = mesh.contains(ligand_coords)
        inside = np.any(inside_mask)

        # Compute signed distances
        signed_distances = mesh.nearest.signed_distance(ligand_coords)
        min_distance = float(np.min(np.abs(signed_distances)))

        # Key fix: treat as valid contact if inside OR very close
        in_contact = inside or min_distance <= LIGAND_THRESHOLD

        return in_contact, mesh, min_distance

    except Exception as e:
        logging.warning(f"⚠️ Failed to validate ligand in extruded mesh: {e}")
        return False, None, None

def compute_circumsphere_radius(tetra):
    if tetra.shape[0] != 4:
        return np.inf

    A, B, C, D = tetra
    AB = B - A
    AC = C - A
    AD = D - A

    M = np.vstack([AB, AC, AD])

    try:
        rhs = 0.5 * np.array(
            [
                np.dot(AB, AB),
                np.dot(AC, AC),
                np.dot(AD, AD),
            ]
        )
        center_relative = np.linalg.solve(M, rhs)
        radius = np.linalg.norm(center_relative)
    except np.linalg.LinAlgError:
        radius = np.inf

    return radius

def monte_carlo_pocket_volume(tetra_positions, pocket_indices, num_samples=10000):
    pocket_tetra = tetra_positions[pocket_indices]
    all_points = pocket_tetra.reshape(-1, 3)
    min_bounds = np.min(all_points, axis=0)
    max_bounds = np.max(all_points, axis=0)
    bbox_volume = np.prod(max_bounds - min_bounds)
    random_points = np.random.uniform(
        low=min_bounds, high=max_bounds, size=(num_samples, 3)
    )
    count_inside = 0
    for p in random_points:
        if any(point_in_tetrahedron(p, tetra) for tetra in pocket_tetra):
            count_inside += 1
    return (count_inside / num_samples) * bbox_volume


def monte_carlo_ligand_volume(ligand_coords, num_samples=10000):
    if ligand_coords.shape[0] < 4:
        return 0.0
    min_bounds = np.min(ligand_coords, axis=0)
    max_bounds = np.max(ligand_coords, axis=0)
    bbox_volume = np.prod(max_bounds - min_bounds)
    delaunay = Delaunay(ligand_coords)
    random_points = np.random.uniform(
        low=min_bounds, high=max_bounds, size=(num_samples, 3)
    )
    inside = delaunay.find_simplex(random_points) >= 0
    return float(np.sum(inside)) / num_samples * bbox_volume


def point_in_tetrahedron(p, tetra):
    A, B, C, D = tetra
    M = np.column_stack([B - A, C - A, D - A])
    try:
        u, v, w = np.linalg.solve(M, p - A)
        return (u >= 0) and (v >= 0) and (w >= 0) and (u + v + w <= 1)
    except np.linalg.LinAlgError:
        return False


def compute_triangle_area(triangle):
    a, b, c = triangle
    return 0.5 * np.linalg.norm(np.cross(b - a, c - a))


def compute_triangle_perimeter(triangle):
    a, b, c = triangle
    return np.linalg.norm(a - b) + np.linalg.norm(b - c) + np.linalg.norm(c - a)


def compute_circumcenter(tetra):
    A, B, C, D = tetra

    M = np.vstack([B - A, C - A, D - A])

    rhs = 0.5 * np.array(
        [
            np.dot(B - A, B - A),
            np.dot(C - A, C - A),
            np.dot(D - A, D - A),
        ]
    )

    try:
        center_relative = np.linalg.solve(M, rhs)
        circumcenter = A + center_relative
    except np.linalg.LinAlgError:
        circumcenter = np.array([np.nan, np.nan, np.nan])

    return circumcenter

def compute_volume_for_tetra_group(tetra_positions, tetra_indices):
    total_volume = 0.0
    for idx in tetra_indices:
        verts = tetra_positions[idx]
        a, b, c, d = verts
        vol = np.abs(np.dot(a - d, np.cross(b - d, c - d))) / 6.0
        total_volume += vol
    return total_volume

def compute_power_radii(tetra_indices, coords, radii):
    
    import numpy as np
    
    tetra_indices = np.asarray(tetra_indices, dtype=int)
    coords = np.asarray(coords, dtype=float)
    radii = np.asarray(radii, dtype=float)
    
    if tetra_indices.ndim != 2 or tetra_indices.shape[1] != 4:
        raise ValueError("tetra_indices deve essere un array (N, 4)")
    
    n_tetra = len(tetra_indices)
    power_radii = np.zeros(n_tetra, dtype=float)
    weights = radii ** 2
    
    for i in range(n_tetra):
        idx = tetra_indices[i]
        P = coords[idx]          # (4, 3)
        w = weights[idx]         # (4,)
        
        p1 = P[0]
        norm2_P = np.sum(P ** 2, axis=1)  # (4,)
        
        # Sistema: 2 (p_j - p_1) · c = ||p_j||^2 - w_j - (||p_1||^2 - w_1), j=2,3,4
        A = 2 * (P[1:] - p1)              # (3, 3)
        b = (norm2_P[1:] - w[1:]) - (norm2_P[0] - w[0])  # (3,)
        
        try:
            c = np.linalg.solve(A, b)
        except np.linalg.LinAlgError:
            # Tetraedro degenere: fallback a circumradius euclideo
            from .geometry_utils import compute_circumsphere_radius
            # Ricostruisci le posizioni dei 4 vertici
            P_vertices = coords[idx]
            power_radii[i] = compute_circumsphere_radius(P_vertices)
            continue
        
        r2 = np.sum((c - p1) ** 2) - w[0]
        if r2 < 0:
            r2 = 0.0
        
        power_radii[i] = np.sqrt(r2)
    
    return power_radii

def compute_ligand_contact_distance(
    protein_coords, ligand_coords, threshold=LIGAND_THRESHOLD
):
    if protein_coords.size == 0 or ligand_coords.size == 0:
        return {}
    distances = cdist(protein_coords, ligand_coords)
    contacting = {}
    for i in range(protein_coords.shape[0]):
        min_dist = distances[i].min()
        if min_dist < threshold:
            contacting[i] = min_dist
    return contacting


def compute_mouth_parameters(
    ranked_pockets, simplices, protein_coords, ligand_coords=None
):
    ligand_center = None
    if ligand_coords is not None and len(ligand_coords) > 0:
        ligand_center = np.mean(ligand_coords, axis=0)
    mouth_params_list = []
    for pocket in ranked_pockets:
        face_count = {}
        for tetra_idx in pocket:
            atoms = simplices[tetra_idx]
            faces = [
                tuple(sorted((atoms[0], atoms[1], atoms[2]))),
                tuple(sorted((atoms[0], atoms[1], atoms[3]))),
                tuple(sorted((atoms[0], atoms[2], atoms[3]))),
                tuple(sorted((atoms[1], atoms[2], atoms[3]))),
            ]
            for face in faces:
                face_count[face] = face_count.get(face, 0) + 1
        boundary_faces = [face for face, count in face_count.items() if count == 1]
        total_area, total_circumference, rim_atoms = 0, 0, set()
        face_centroids = []
        

        for face in boundary_faces:
            pts = protein_coords[list(face)]
            area = compute_triangle_area(pts)
            total_area += area
            centroid = np.mean(pts, axis=0)
            face_centroids.append(centroid)
            edges = [(face[i], face[j]) for i, j in [(0, 1), (1, 2), (2, 0)]]
            total_circumference += sum(
                np.linalg.norm(protein_coords[e[0]] - protein_coords[e[1]])
                for e in edges
            )
            rim_atoms.update(face)
        mouth_center = (
            np.mean(face_centroids, axis=0) if face_centroids else np.zeros(3)
        )
        if ligand_coords is not None and len(ligand_coords) > 0:
            dists = np.linalg.norm(ligand_coords - mouth_center[None, :], axis=-1)
            ligand_distance = float(np.min(dists))
        else:
            ligand_distance = None
        mouth_params_list.append(
            {
                "mouth_area": round(total_area, 2),
                "mouth_circumference": round(total_circumference, 2),
                "rim_atoms": list(rim_atoms),
                "mouth_center": mouth_center.tolist(),
                "ligand_distance": (
                    round(ligand_distance, 2) if ligand_distance is not None else None
                ),
            }
        )
    return mouth_params_list

def merge_nearby_clusters(pockets, tetra_positions, threshold):
    if not pockets:
        return []
    clusters = list(pockets)
    changed = True
    while changed:
        changed = False
        new_clusters = []
        merged_flags = [False] * len(clusters)
        for i in range(len(clusters)):
            if merged_flags[i]:
                continue
            cluster_i = clusters[i]
            if not cluster_i:
                continue
            tetra_coords_i = np.concatenate(
                [tetra_positions[idx] for idx in cluster_i], axis=0
            )
            centroid_i = np.mean(tetra_coords_i, axis=0)
            for j in range(i + 1, len(clusters)):
                if merged_flags[j]:
                    continue
                cluster_j = clusters[j]
                if not cluster_j:
                    continue
                tetra_coords_j = np.concatenate(
                    [tetra_positions[idx] for idx in cluster_j], axis=0
                )
                centroid_j = np.mean(tetra_coords_j, axis=0)
                if np.linalg.norm(centroid_i - centroid_j) < threshold:
                    cluster_i = list(set(cluster_i) | set(cluster_j))
                    merged_flags[j] = True
                    changed = True
                    tetra_coords_i = np.concatenate(
                        [tetra_positions[idx] for idx in cluster_i], axis=0
                    )
                    centroid_i = np.mean(tetra_coords_i, axis=0)
            new_clusters.append(cluster_i)
        clusters = new_clusters
    return clusters

def apply_splitting_to_shared_atoms(pockets, protein_coords, atom_radii):
    if not ATOM_SPLITTING:
        return pockets
    for i in range(len(pockets)):
        for j in range(i + 1, len(pockets)):
            pocket_i = set(pockets[i])
            pocket_j = set(pockets[j])
            common_atoms = pocket_i.intersection(pocket_j)
            if common_atoms:
                pocket_center1 = np.mean(protein_coords[list(pocket_i)], axis=0)
                pocket_center2 = np.mean(protein_coords[list(pocket_j)], axis=0)
                for atom in common_atoms:
                    atom_coord = protein_coords[atom]
                    atom_radius = atom_radii[atom] if atom_radii is not None else 1.0
                    if ATOM_SPLITTING_MODE.upper() == "CAST":
                        frac1, frac2 = split_atom_contribution(
                            atom_coord, atom_radius, pocket_center1, pocket_center2
                        )
                    else:
                        frac1, frac2 = 0.5, 0.5
                    logging.info(
                        f"Atom {atom} shared between pockets {i} and {j}: split fractions {frac1:.2f}, {frac2:.2f}"
                    )
    return pockets


def split_atom_contribution(atom_coord, atom_radius, pocket_center1, pocket_center2):
    midpoint = (pocket_center1 + pocket_center2) / 2
    plane_normal = pocket_center2 - pocket_center1
    norm = np.linalg.norm(plane_normal)
    if norm == 0:
        return (0.5, 0.5)
    plane_normal = plane_normal / norm
    d = np.abs(np.dot(atom_coord - midpoint, plane_normal))
    frac = compute_spherical_cap_fraction(atom_radius, d)
    signed_d = np.dot(atom_coord - midpoint, plane_normal)
    if signed_d >= 0:
        return (1 - frac, frac)
    else:
        return (frac, 1 - frac)


def compute_spherical_cap_fraction(R, d):
    if d >= R:
        return 0
    V_cap = (np.pi * (R - d) ** 2 * (2 * R + d)) / 3.0
    V_sphere = (4 / 3) * np.pi * R**3
    return V_cap / V_sphere

def load_and_separate_pdb(pdb_path, include_water=INCLUDE_WATER):
    ppdb = PandasPdb().read_pdb(pdb_path)

    # --- Protein coordinates ---
    protein_df = ppdb.df["ATOM"]
    protein_coords = protein_df[["x_coord", "y_coord", "z_coord"]].values
    logging.info(f"Protein coordinates loaded: {protein_coords.shape[0]} atoms.")

    # --- Ligand coordinates ---
    ligand_df = ppdb.df.get("HETATM")
    if ligand_df is not None and not ligand_df.empty:
        col_name = "residue_name" if "residue_name" in ligand_df.columns else "blank_1"
        if not include_water:
            ligand_df = ligand_df[ligand_df[col_name] != "HOH"]
        ligand_coords = (
            ligand_df[["x_coord", "y_coord", "z_coord"]].values
            if not ligand_df.empty
            else np.array([])
        )
    else:
        ligand_coords = np.array([])
    logging.info(f"Ligand coordinates loaded: {ligand_coords.shape[0]} atoms.")

    # --- Build list of protein atoms with metadata ---
    protein_atoms = []
    for _, row in protein_df.iterrows():
        protein_atoms.append(
            {
                "resid": int(row["residue_number"]),
                "atom_name": row["atom_name"].strip(),
                "coord": [row["x_coord"], row["y_coord"], row["z_coord"]],
            }
        )

    return protein_coords, ligand_coords, protein_atoms, ppdb


def calculate_atomic_radii(ppdb):
    try:
        protein_df = ppdb.df.get("ATOM")
        if protein_df is None or protein_df.empty:
            raise ValueError("No ATOM data available to compute radii.")
        atom_elements = protein_df["element_symbol"].values
        radii = np.array(
            [ATOMIC_RADII_DICT.get(elem, DEFAULT_RADIUS) for elem in atom_elements]
        )
        logging.info(f"Calculated atomic radii for {len(protein_df)} atoms.")
    except Exception as e:
        logging.error(f"Error calculating atomic radii: {e}")
        raise
    return radii


def extract_connectivity(ppdb, distance_threshold=CONNECTIVITY_THRESHOLD):
    connectivity = {}
    if "CONECT" in ppdb.df and not ppdb.df["CONECT"].empty:
        logging.info("Using CONECT records to extract connectivity.")
        conect_df = ppdb.df["CONECT"]
        for idx, row in conect_df.iterrows():
            source = int(row[0]) - 1  # 0-based index
            neighbors = [int(x) - 1 for x in row[1:] if not pd.isna(x)]
            connectivity[source] = neighbors
        num_atoms = len(ppdb.df["ATOM"])
        for i in range(num_atoms):
            if i not in connectivity:
                connectivity[i] = []
    else:
        logging.info(
            "No CONECT records found; computing connectivity based on distance threshold."
        )
        protein_df = ppdb.df.get("ATOM")
        if protein_df is None or protein_df.empty:
            raise ValueError("No ATOM data available to compute connectivity.")
        coords = protein_df[["x_coord", "y_coord", "z_coord"]].to_numpy()
        tree = KDTree(coords)
        num_atoms = coords.shape[0]
        for i in range(num_atoms):
            indices = tree.query_ball_point(coords[i], r=distance_threshold)
            indices = [j for j in indices if j != i]
            connectivity[i] = indices
    return connectivity

def load_alpha_artifacts(results_dir):
    results = Path(results_dir)
    csvs = sorted(results.glob('*alphasweepmetrics.csv'))

    if not csvs or not jsons:
        raise FileNotFoundError('Alpha diagnostics artifacts not found in results_dir')
    csv_path = csvs[0]
    json_path = jsons[0]
    df = pd.read_csv(csv_path)
    with open(json_path, 'r', encoding='utf-8') as f:

def recommend_alpha_from_sweep(df):
    work = df.copy()
    required = ['alpha', 'numfinalpockets', 'maxvolume', 'meanflowsteps']
    missing = [c for c in required if c not in work.columns]
    if missing:
        raise ValueError(f'Missing columns for alpha recommendation: {missing}')
    for col in required:
        work[col] = pd.to_numeric(work[col], errors='coerce').fillna(0)
    work = work.sort_values('alpha').reset_index(drop=True)
    work['vol_norm'] = work['maxvolume'] / max(work['maxvolume'].max(), 1.0)
    work['pocket_norm'] = work['numfinalpockets'] / max(work['numfinalpockets'].max(), 1.0)
    work['flow_norm'] = work['meanflowsteps'] / max(work['meanflowsteps'].max(), 1.0)
    work['delta_volume'] = work['maxvolume'].diff().abs().fillna(0) + work['maxvolume'].diff(-1).abs().fillna(0)
    work['stability'] = 1.0 / (1.0 + work['delta_volume'])
    work['score'] = 0.45 * work['vol_norm'] + 0.20 * work['flow_norm'] + 0.25 * work['stability'] - 0.10 * work['pocket_norm']
    best_idx = work['score'].idxmax()
    best_row = work.loc[best_idx].to_dict()
    return {
        'recommended_alpha': float(best_row['alpha']),
        'score': float(best_row['score']),
        'numfinalpockets': int(best_row['numfinalpockets']),
        'maxvolume': float(best_row['maxvolume']),
        'meanflowsteps': float(best_row['meanflowsteps']),
    }

def recommend_alpha_from_connectivity_sweep(df):
    work = df.copy()

    required = [
        "alpha",
        "alpha_fraction",
        "largest_component_ratio",
        "small_component_fraction",
        "num_alpha_components",
    ]
    missing = [c for c in required if c not in work.columns]
    if missing:
        raise ValueError(f"Missing columns for connectivity alpha recommendation: {missing}")

    for col in required:
        work[col] = pd.to_numeric(work[col], errors="coerce").fillna(0)

    work = work.sort_values("alpha").reset_index(drop=True)

    work["connectivity_score"] = (
        0.60 * work["largest_component_ratio"]
        - 0.20 * work["small_component_fraction"]
        - 0.10 * (work["alpha_fraction"] / max(work["alpha_fraction"].max(), 1e-12))
        - 0.10 * (work["num_alpha_components"] / max(work["num_alpha_components"].max(), 1))
    )

    best_idx = work["connectivity_score"].idxmax()
    best_row = work.loc[best_idx].to_dict()

    return {
        "recommended_alpha": float(best_row["alpha"]),
        "connectivity_score": float(best_row["connectivity_score"]),
        "largest_component_ratio": float(best_row["largest_component_ratio"]),

def compute_sasa(pdbpath, method=None, remove_ligand=False):
    structure = freesasa.Structure(pdbpath)
    result = freesasa.calc(structure)
    sasa_values = {}
    for i in range(structure.nAtoms()):
        resi = structure.residueNumber(i)
        area = result.atomArea(i)
        sasa_values[resi] = sasa_values.get(resi, 0) + area
    return sasa_values


def compute_ligand_contact_sasa(sasa_before, sasa_after):
    if not sasa_before or not sasa_after:
        logging.warning("Incomplete SASA data; cannot compute ligand contact.")
        return {}
    contacting = {}
    for atom_idx, area_before in sasa_before.items():
        area_after = sasa_after.get(atom_idx, 0)
        if area_before > area_after:
            contacting[atom_idx] = area_before
    return contacting


def evaluate_sasa_contact(
    pocket_atom_indices,
    sasa_before,
    sasa_after,
    ligand_coords,
    protein_coords,
    ):
    contact_atoms = []
    for idx in pocket_atom_indices:
        res_id = str(idx)
        sasa_b = sasa_before.get(res_id, 0)
        sasa_a = sasa_after.get(res_id, 0)
        if sasa_b > sasa_a:
            contact_atoms.append(protein_coords[idx])
    if not contact_atoms or ligand_coords.size == 0:
        return False
    distances = cdist(np.array(contact_atoms), ligand_coords)
    min_dist = np.min(distances)
    return min_dist <= threshold

def _tet_edges(tets):

            lines += [2, idx, idx + 1]
            idx += 2

    mesh = pv.PolyData()
    mesh.points = np.array(pts, dtype=np.float32)
    mesh.lines = np.array(lines, dtype=np.int32)

def guess_alpha_values(mol: MolData) -> Dict[str, float]:
    from scipy.spatial import KDTree

    r_mean = float(np.mean(mol.radii))
    tree = KDTree(mol.coords)
    dists, _ = tree.query(mol.coords, k=5)
    d_nn = float(np.mean(dists[:, 1:5]))
    centroid = mol.coords.mean(axis=0)
    rg = float(np.sqrt(np.mean(np.sum((mol.coords - centroid) ** 2, axis=1))))

    return {
        "tight": round(r_mean * 1.00, 2),
        "medium": round(r_mean * 1.40, 2),
        "loose": round(r_mean * 1.95, 2),
        "_r_mean": r_mean,
        "_d_nn": d_nn,
        "_rg": rg,
    }

"""
Minimal occlusion viewer for PyCASTA.
Shows protein atoms colored by occlusion fraction, with ligand overlaid.
"""

import json
import os
import sys
import numpy as np
import plotly.graph_objects as go
from pathlib import Path


def load_pdb_atoms(pdb_file):
    """Load atoms from PDB file (simple parser)."""
    atoms = []
    with open(pdb_file, 'r') as f:
        for line in f:
            if line.startswith('ATOM') or line.startswith('HETATM'):
                x = float(line[30:38])
                y = float(line[38:46])
                z = float(line[46:54])
                atom_name = line[12:16].strip()
                residue = line[17:20].strip()
                chain = line[21]
                resid = int(line[22:26])
                element = line[76:78].strip() if len(line) > 76 else atom_name[0]
                is_hetero = line.startswith('HETATM')
                atoms.append({
                    'coord': [x, y, z],
                    'atom_name': atom_name,
                    'residue': residue,
                    'chain': chain,
                    'resid': resid,
                    'element': element,
                    'is_hetero': is_hetero,
                })
    return atoms


def compute_occlusion_for_atoms(atoms, num_rays=64, max_distance=15.0):
    """
    Compute occlusion fraction for each atom.
    Occlusion = fraction of rays that hit other atoms.
    """
    coords = np.array([a['coord'] for a in atoms], dtype=float)
    n_atoms = len(atoms)
    occlusion_values = np.zeros(n_atoms)
    
    # Simple neighbor-based occlusion (faster than full raycasting for visualization)
    from scipy.spatial import KDTree
    tree = KDTree(coords)
    
    for i in range(n_atoms):
        center = coords[i]
        # Find neighbors within max_distance
        neighbors = tree.query_ball_point(center, r=max_distance)
        # Occlusion ~ fraction of solid angle covered by neighbors
        # Simplified: count neighbors and their relative sizes
        if len(neighbors) > 1:
            # More neighbors = more occluded
            occlusion_values[i] = min(1.0, (len(neighbors) - 1) / 20.0)  # Normalize
    
    return occlusion_values


def make_occlusion_figure(pdb_file, ligand_residues=None, output_html=None):
    """
    Create an interactive HTML visualization of occlusion.
    
    Parameters
    ----------
    pdb_file : str
        Path to PDB file.
    ligand_residues : list, optional
        List of residue names/IDs that are ligands (e.g., ['ATP', 'HEM']).
        If None, all HETATM are considered ligands.
    output_html : str, optional
        Output HTML file path. If None, returns figure object.
    """
    # Load atoms
    atoms = load_pdb_atoms(pdb_file)
    coords = np.array([a['coord'] for a in atoms], dtype=float)
    
    # Separate protein and ligand
    protein_atoms = [a for a in atoms if not a['is_hetero']]
    ligand_atoms = [a for a in atoms if a['is_hetero']]
    
    if ligand_residues is not None:
        ligand_atoms = [a for a in ligand_atoms if a['residue'] in ligand_residues]
    
    protein_coords = np.array([a['coord'] for a in protein_atoms], dtype=float)
    ligand_coords = np.array([a['coord'] for a in ligand_atoms], dtype=float)
    
    # Compute occlusion for protein atoms
    if len(protein_coords) > 0:
        occlusion_values = compute_occlusion_for_atoms(protein_atoms)
    else:
        occlusion_values = np.array([])
    
    # Create figure
    fig = go.Figure()
    
    # Protein atoms (colored by occlusion)
    if len(protein_coords) > 0:
        fig.add_trace(go.Scatter3d(
            x=protein_coords[:, 0],
            y=protein_coords[:, 1],
            z=protein_coords[:, 2],
            mode='markers',
            marker=dict(
                size=3,
                color=occlusion_values,
                colorscale='RdYlBu_r',  # Red = occluded, Blue = exposed
                colorbar=dict(title='Occlusion'),
                opacity=0.8,
            ),
            name='Protein (occlusion)',
            text=[f"{a['residue']} {a['resid']}" for a in protein_atoms],
        ))
    
    # Ligand atoms (as spheres)
    if len(ligand_coords) > 0:
        fig.add_trace(go.Scatter3d(
            x=ligand_coords[:, 0],
            y=ligand_coords[:, 1],
            z=ligand_coords[:, 2],
            mode='markers',
            marker=dict(
                size=6,
                color='red',
                opacity=0.9,
            ),
            name='Ligand',
        ))
    
    # Layout
    fig.update_layout(
        title=f"Occlusion Viewer - {Path(pdb_file).stem}",
        scene=dict(
            aspectmode='data',
            xaxis_title='X (Å)',
            yaxis_title='Y (Å)',
            zaxis_title='Z (Å)',
        ),
        margin=dict(l=0, r=0, t=40, b=0),
        height=700,
    )
    
    # Save or return
    if output_html:
        fig.write_html(output_html, include_plotlyjs='cdn')
        print(f"Occlusion viewer saved to: {output_html}")
        return output_html
    else:
        return fig


def main():
    if len(sys.argv) < 2:
        print("Usage: python tools/occlusion_viewer.py <pdb_file> [output_html]")
        print("Example: python tools/occlusion_viewer.py data/bounded/1a6w.pdb occlusion_1a6w.html")
        sys.exit(1)
    
    pdb_file = sys.argv[1]
    output_html = sys.argv[2] if len(sys.argv) > 2 else None
    
    if output_html is None:
        base_name = Path(pdb_file).stem
        output_html = f"occlusion_{base_name}.html"
    
    make_occlusion_figure(pdb_file, output_html=output_html)


if __name__ == "__main__":
    main()

    p.add_argument("--rays", type=int, default=64)
    p.add_argument("--res", type=float, default=0.5)
    p.add_argument("--results-dir", type=str, default=None)

import sys, argparse, os
import numpy as np

sys.path.insert(0, os.path.dirname(os.path.abspath(__file__)))
from core import MolData, get_vdw, guess_alpha_values
from alpha_viewer import show_alpha


def load_protein_atoms_from_pdb(pdb_path):
    coords = []
    atom_names = []
    with open(pdb_path, encoding="utf-8") as f:
        for line in f:
            if line.startswith("ATOM"):
                atom_name = line[12:16].strip()
                x = float(line[30:38].strip())
                y = float(line[38:46].strip())
                z = float(line[46:54].strip())
                coords.append([x, y, z])
                atom_names.append(atom_name)
    coords = np.array(coords, dtype=float)
    radii = np.array([get_vdw(name) for name in atom_names], dtype=float)
    ligand_coords = np.empty((0, 3), dtype=float)
    representative_points = np.empty((0, 3), dtype=float)
    return coords, atom_names, radii, ligand_coords, representative_points


def load_alpha_only(pdb_path, alpha_npz_path, alpha_value=None):
    coords, atom_names, radii, ligand_coords, representative_points = load_protein_atoms_from_pdb(pdb_path)
    with np.load(alpha_npz_path, allow_pickle=True) as alpha_data:
        tetrahedra = np.array(alpha_data["tetrahedra"], dtype=float)
        alpha_mask = np.array(alpha_data["alpha_mask"], dtype=bool)

    mol = MolData(
        name=os.path.splitext(os.path.basename(pdb_path))[0],
        pdb_path=str(pdb_path),
        coords=coords,
        radii=radii,
        atom_names=atom_names,
        ligand_coords=ligand_coords,
        tetrahedra=tetrahedra,
        alpha_mask=alpha_mask,
        ranked_pockets=[],
        representative_points=representative_points,
        pocket_volumes=[],
        ligand_validation=[],
        ranking_scores=[],
        alpha_used=float(alpha_value) if alpha_value is not None else None,
        ao_scores=None,
        alpha_npz_path=str(alpha_npz_path),
    )
    return mol


def main():
    p = argparse.ArgumentParser(description="PyCASTA alpha-only visualization without JSON")
    p.add_argument("pdb")
    p.add_argument("alpha_npz")
    p.add_argument("--alpha", type=float, default=None)
    args = p.parse_args()

    mol = load_alpha_only(args.pdb, args.alpha_npz, args.alpha)
    av = guess_alpha_values(mol)
    if args.alpha is not None:
        av["custom"] = args.alpha

    print(f"\nMolecola  : {mol.name}")
    print(f"Atomi     : {len(mol.coords)}")
    print(f"Tetraedri : {len(mol.tetrahedra)}  (inside={mol.alpha_mask.sum()}, outside={(~mol.alpha_mask).sum()})")
    print(f"Alpha     : tight={av['tight']:.2f}  medium={av['medium']:.2f}  loose={av['loose']:.2f}")

    show_alpha(mol, av)


if __name__ == "__main__":
    main()

"""visualization/surface_viewer.py - Finestra 2/3: Superficie VdW colorata per AO"""
import numpy as np
import pyvista as pv
from core import MolData



def _build_surface(coords, radii, res=0.5):
    pad = 3.0
    lo = coords.min(0) - pad
    hi = coords.max(0) + pad
    nx, ny, nz = int((hi[0] - lo[0]) / res) + 1, int((hi[1] - lo[1]) / res) + 1, int((hi[2] - lo[2]) / res) + 1
    gx, gy, gz = np.meshgrid(
        np.linspace(lo[0], hi[0], nx),
        np.linspace(lo[1], hi[1], ny),
        np.linspace(lo[2], hi[2], nz),
        indexing="ij",
    )
    gpts = np.stack([gx.ravel(), gy.ravel(), gz.ravel()], axis=1)
    sdf = np.full(len(gpts), -np.inf, dtype=np.float32)
    for s in range(0, len(coords), 64):
        e = min(s + 64, len(coords))
        d = np.linalg.norm(gpts[:, None, :] - coords[None, s:e, :], axis=2)
        sdf = np.maximum(sdf, (radii[None, s:e] - d).max(axis=1))
    grid = pv.ImageData(dimensions=(nx, ny, nz), spacing=(res, res, res), origin=lo)
    grid.point_data["sdf"] = sdf.reshape(nx, ny, nz).ravel(order="F")
    return grid.contour([0.0], scalars="sdf")


def _interp_ao(surface, coords, ao, k=4):
    from scipy.spatial import KDTree
    dists, idxs = KDTree(coords).query(np.array(surface.points), k=k)
    dists = np.clip(dists, 1e-6, None)
    w = 1 / dists**2
    w /= w.sum(1, keepdims=True)
    return (w * ao[idxs]).sum(1).astype(np.float32)


def _toggle(actor, plotter):
    if actor is not None:
        actor.SetVisibility(not bool(actor.GetVisibility()))
        plotter.render()


def show_surface(mol: MolData, N_rays=64, resolution=0.5):
    print(f"\n=== Surface AO: {mol.name} ===")
    if mol.ao_scores is None:
        print(f" Calcolo AO ({N_rays} raggi)...")
        mol.ao_scores = compute_ao(mol.coords, mol.radii, N_rays=N_rays)
    ao = mol.ao_scores
    print(f" AO min={ao.min():.3f} max={ao.max():.3f} mean={ao.mean():.3f}")
    print(f" Costruzione superficie ({resolution} A)...")
    surf = _build_surface(mol.coords, mol.radii, resolution)
    surf.point_data["AO"] = _interp_ao(surf, mol.coords, ao)
    print(f" {surf.n_points} vertici, {surf.n_cells} celle")

    pl = pv.Plotter(window_size=(1000, 800), title=f"Surface AO - {mol.name}")
    pl.set_background("#0f1117")
    actors = {}

    actors["surface"] = pl.add_mesh(
        surf,
        scalars="AO",
        cmap="RdBu_r",
        clim=(0, 1),
        smooth_shading=True,
        scalar_bar_args={
            "title": "Occlusion\n(0=esposto 1=cavita)",
            "color": "white",
            "title_font_size": 12,
            "label_font_size": 10,
        },
    )

    if len(mol.ligand_coords):
        actors["ligand"] = pl.add_mesh(
            pv.PolyData(mol.ligand_coords.astype(np.float32)),
            color="yellow",
            point_size=14,
            render_points_as_spheres=True,
            opacity=0.95,
            label="Ligand",
        )

    reps = np.asarray(mol.representative_points, dtype=np.float32) if len(mol.representative_points) else np.empty((0, 3), dtype=np.float32)
    actors["rep_points"] = None
    if len(reps):
        actors["rep_points"] = pl.add_mesh(
            pv.PolyData(reps),
            color="#ff4d6d",
            point_size=13,
            render_points_as_spheres=True,
            opacity=1.0,

        )

    pl.add_key_event("1", lambda: _toggle(actors.get("surface"), pl))
    pl.add_key_event("2", lambda: _toggle(actors.get("ligand"), pl))
    pl.add_key_event("3", lambda: _toggle(actors.get("rep_points"), pl))
    pl.add_key_event("0", pl.reset_camera)

    pl.add_text(
        "1 surface | 2 ligand | 3 rep points | 0 reset",
        position="upper_left",
        font_size=10,
        color="white",
    )
    pl.show()

# -*- coding: utf-8 -*-
"""visualization/triangulation_viewer.py"""

import numpy as np
import pyvista as pv


def _tetra_edges():
    return [
        (0, 1), (0, 2), (0, 3),
        (1, 2), (1, 3), (2, 3),
    ]


def _build_wireframe_lines(tetrahedra: np.ndarray):
    points = []
    lines = []

    for tet in tetrahedra:
        base = len(points)
        points.extend(tet.tolist())

        for i, j in _tetra_edges():
            lines.append(2)
            lines.append(base + i)
            lines.append(base + j)

    if not points:
        return None

    poly = pv.PolyData(np.asarray(points, dtype=float))
    poly.lines = np.asarray(lines, dtype=np.int32)
    return poly


def show_triangulation(mol, max_tetra=None):
    tetra = getattr(mol, "triangulation_tetrahedra", None)

    if tetra is None or len(tetra) == 0:
        print("[triangulation_viewer] No triangulation debug data available.")
        return

    tetra = np.asarray(tetra, dtype=float)

    if max_tetra is not None and len(tetra) > max_tetra:
        tetra = tetra[:max_tetra]

    line_mesh = _build_wireframe_lines(tetra)
    if line_mesh is None:
        print("[triangulation_viewer] Empty triangulation.")
        return

    pl = pv.Plotter(window_size=(1400, 900))
    pl.set_background("#0b0f14")

    if getattr(mol, "coords", None) is not None and len(mol.coords) > 0:
        pl.add_points(
            np.asarray(mol.coords, dtype=float),
            color="white",
            point_size=4,
            render_points_as_spheres=True,
            opacity=0.35,
        )

    if getattr(mol, "ligand_coords", None) is not None and len(mol.ligand_coords) > 0:
        pl.add_points(
            np.asarray(mol.ligand_coords, dtype=float),
            color="#ff7f50",
            point_size=9,
            render_points_as_spheres=True,
            opacity=1.0,
        )

    pl.add_mesh(
        line_mesh,
        color="#9aa4b2",
        line_width=1,
        opacity=0.35,
    )

    pl.add_text(
        f"{mol.name} | raw triangulation | tetra={len(tetra)}",
        font_size=12,
        color="white",
    )
    pl.add_axes()
    pl.show()
